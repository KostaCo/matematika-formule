\chapter{Logika}
\section{Relacije}
Neka je $\rho$ binarna relacija skupa $A$. Tada, ona može imati sledeće osobine:\\
\begin{itemize}
    \item \textbf{Refleksivnost}: $x \; \rho \; x, \;\forall x \in A$\\
    \item \textbf{Simetričnost}: $x \; \rho \; y \implies y \; \rho \; x, \;\forall x,y \in A$\\
    \item \textbf{Antisimetričnost}: $x \; \rho \; y \land y \; \rho \; x \implies x = y, \;\forall x,y \in A$\\
    \item \textbf{Tranzitivnost}: $x \;\rho\; y \;\land \; y \;\rho\; z \implies x \;\rho \; z, \; \forall x,y,z \in A$
\end{itemize}
\subsection{Relacije ekvivalencije i poretka}
\begin{itemize}
    \item \textbf{Ekvivalencija}: refleksivnost + simetričnost + tranzitivnost
    \item \textbf{Poredak}: refleksivnost + antisimetričnost + tranzitivnost
\end{itemize}
\subsection{\textit{1-1} i \textit{na} funkcije}
\begin{itemize}
    \item \textbf{1-1}: $x_1=x_2 \implies f(x_1) = f(x_2)$
    \item \textbf{na}: $f:A\xrightarrow{} B, \; (\forall y \in B)(\exists x \in A)(y = f(x))$
\end{itemize}
\section{Operacije}
\begin{itemize}
    \item \textbf{Komutativnost}: $x+y=y+x$
    \item \textbf{Asocijativnost}: $x+(y+z) = (x+y)+z$
    \item \textbf{Distributivnost ($\cdot$ prema $+$ sa desne strane)}: $(x+y)\cdot z = (x \cdot z) + (y \cdot z)$
    \item \textbf{Distributivnost ($\cdot$ prema $+$ sa leve strane)}: $z \cdot (x+y) = (z \cdot x) + (z \cdot y)$
\end{itemize}

\section{Polinomi}
\subsection{Prsten polinoma}
\begin{itemize}
    \item \textbf{Polinom} nad poljem $\mathbb{F}$:
    \[ f(x) = \alpha_nx^n + \alpha_{n-1}x^{n-1}+ \dots + \alpha_1x + \alpha_0, \;\;\; \alpha_i \in \mathbb{F},\; \alpha_n \neq 0 \]
    \item $\deg f = \partial f = n$ - \textbf{stepen polinoma}
    \item Ne-nula polinom stepena $n$ je \textbf{moničan} ako je $\alpha_n = 1$.
    \item $\mathcal{P} = \mathbb{F}[x]$ - skup svih polinoma nad $\mathbb{F}$
    \item $\mathbb{F}[x] = (\mathbb{F}[x], \mathbb{F}, +, \cdot)$ je vektorski prostor.
    \item $(\mathbb{F}, +, \cdot)$ je komutativan i asocijativan prsten sa jedinicom. Skup invertibilinih elemenata je $\mathbb{F}^* = 
    \mathbb{F}\setminus \{0\}$ (svi ne-nula skalari).
    \item $\mathbb{F}[x]$ je asocijativna i komutativna $\mathbb{F}$-algebra sa jedinicom.
\end{itemize}

\begin{theorem}
    Neka su $0 \neq f, g, h \in \mathbb{F}[x]$. Tada:
    \begin{itemize}
        \item $fg \neq 0$
        \item $\partial(fg) = \partial f + \partial g$
        \item $fg$ je moničan ako su $f$ i $g$ monični
        \item $fg$ je skalar $\iff$ $f$ i $g$ su skalari
        \item $\partial(f + g) \neq 0 \implies \partial(f + g) \leq \max (\partial f, \partial g)$
        \item $fg = fh \implies g = h$
    \end{itemize}
\end{theorem}

\subsection{Teorema o deljenju polinoma}
\begin{theorem}
    \textbf{(O deljenju polinoma):} Za svaka dva polinoma $f, g \in \mathbb{F}[x]$ postoje jedinstveni $q, r \in \mathbb{F}[x]$ takvi da:
    \begin{itemize}
        \item $f(x) = q(x)g(x) + r(x)$
        \item $0 \leq \partial r < \partial g$
    \end{itemize}
\end{theorem}

\begin{lemma}
    \textbf{(Bezuova):} Neka je $f \in \mathbb{F}[x]$ polinom. Tada važi
    \[ f(a) = 0 \iff x - a \;|\; f(x) \]
\end{lemma}

\begin{itemize}
    \item Polinom $g \in \mathbb{F}[x]$ je \textbf{delilac} polinoma $f \in \mathbb{F}[x]$ akko postoji polinom $h \in \mathbb{F}$ takav da je $f = gh$ (pišemo $g \; | \; f$).
\end{itemize}
\newpage
\begin{theorem}
    Relacija deljivosti polinoma ima sledeće osobine:
    \begin{enumerate}
        \item $g \; | \; f \land h \; | \; g \implies h \; | \; f$ (tranzitivnost)
        \item $g \; | \; f \land g \; | \; h \implies g \; | \; f \pm h$
        \item $g \; | \; f \implies g \; | \; fh, \; \forall h \in \mathbb{F}[x]$
        \item $g \; | \; f_1, \dots, f_k \implies g \; | \; f_1g_1 + \dots + f_kg_k, \; \forall g_i \in \mathbb{F}[x]$
        \item $g(x) = c \in \mathbb{F}^* \implies g \; | \; f, \; \forall f \in \mathbb{F}[x]$
        \item $g \; | \; f \implies cg \; | \; f, \; \forall c \in \mathbb{F}$
        \item $g \; | \; f \land \partial g = \partial f \iff g = cf, \; c \in \mathbb{F}^*$
    \end{enumerate}
\end{theorem}\

\begin{itemize}
    \item Polinom $d(x)$ je \textbf{najveći zajednički delilac} $\operatorname{NZD}(f(x), g(x))$ polinoma $f(x)$ i $g(x)$ akko $d(x)$ ima sledeće osobine:
    \begin{description}
        \item[(NZD1)] $d\;|\;f \land d\;|\;g$
        \item[(NZD2)] $d_1\;|\;f \land d_1\;|\;g \implies d_1\;|\;d$
        \item[(NZD3)] vodeći koeficijent polinoma $d$ je 1
    \end{description}
    \item Ako je $\operatorname{NZD}(f, g)=1$, kažemo da su $f$ i $g$ uzajamno prosti.
\end{itemize}

\subsection{Posledice teoreme o deljenju polinoma}

\begin{theorem}
    \textbf{(Euklidov algoritam):} Neka su $f, g \in \mathbb{F}[x]$ polinomi takvi da je  $\deg f \geq \deg g$. Tada je dobro
    definisan konačan niz jednakosti:
    \[
        \begin{aligned}
            &f = gq_1 + r1,\\
            &g = r_1q_2 + r_2, \\
            &r_1 = r_2q_3 + r_3, \\
            &\vdots\\
            &r_{k-3} = r_{k-2}q_{k-1} + r_{k-1},\\
            &r_{k-2} = r_{k-1}q_k + r_k,\\
            &r_{k-1} = r_kq_{k+1}.
        \end{aligned}
    \]
    Tada je $\operatorname{NZD}(f, g)=cr_k$, gde je $c \in \mathbb{F}^*$ jednak recipročnoj vrednosti vodećeg koeficijenta
    polinoma $r_k$.
\end{theorem}

\begin{theorem}
    Neka su $f, g \in \mathbb{F}[x]$ polinomi i neka je $\operatorname{NZD}(f,g)=d$. Tada postoje polinomi $u, v \in \mathbb{F}[x]$
    takvi da je
    \[ fu + gv = d. \]
    Ako je $\deg f > 0$ i $\deg g > 0$, onda možemo odabrati $u$ i $v$ td. važi $\deg u < \deg g$ i $\deg v < \deg f$.
\end{theorem}
\noindent\underline{\textit{Intuicija:}} Ovo važi jer "namestimo" $u$ i $v$ tako da se članovi najvećeg stepena skrate. Na primer:
\[ f(x) = 3x^4 + 5x^3 + x^2 + 5x - 2, \;\;\; g(x) = 3x^4 + 11x^3 + 23x^2 + 21x - 10 \]
\[ \implies \operatorname{NZD}(f, g) = x^2 + \frac{5}{3}x - \frac{2}{3} \]
\[ fu + gv = \operatorname{NZD}(f, g) \implies u(x) = \frac{x}{30}, \;\; v(x) = \frac{-x + 2}{30} \]

\subsection{Nule polinoma}

\begin{theorem}
    \textbf{(O kompleksnim nulama realnog polinoma):} Neka je $f \in \mathbb{R}[x]$ i $z \in \mathbb{C}\setminus\mathbb{R}$. Tada važi
    \[ f(z) = 0 \implies f(\overline{z}) = 0. \]
\end{theorem}

\subsection{Ireducibilnost(nesvodljivost) polinoma}
\begin{itemize}
    \item Polinom $f(x) \in \mathbb{F}[x]$ je \textbf{reducibilan (svodljiv)} nad $\mathbb{F}$ ako postoje polinomi $g, h \in 
    \mathbb{F}[x]$ td. važi:
    \begin{enumerate}
        \item $f(x) = g(x)h(x)$,
        \item $\deg g \geq 1 \land \deg g \geq 1$.
    \end{enumerate}
    Ako takvi polinomi ne postoje, onda je \textbf{ireducibilan (nesvodljiv)}.
    \item Ireducibilni polinom stepena barem 1 nazivamo \textbf{prostim polinomom} nad $\mathbb{F}$.
\end{itemize}

\begin{theorem}
    Neka $p, f, g \in \mathbb{F}[x]$. Tada važi: 
    \[p \text{ je prost polinom } \land\; p \;|\; fg \implies p \;|\; f \lor p \;|\; g\]
\end{theorem}

\noindent\textbf{Posledica 1.} \textit{Ako je $p \in \mathbb{F}[x]$ prost polinom i $p\;|\; f_1\dots f_m$, tada $p$ deli
barem jedan od polinoma $f_i$.}

\begin{theorem}
    Neka je $f \in \mathbb{F}[x]$ i $\deg f \geq 1$. Tada postoji $a_n \in \mathbb{F}$ i različiti prosti monični polinomi 
    $p_1, \dots, p_m$ i $r_1, \dots, r_k \in \mathbb{N}$ td. važi
    \[ f(x) = a_np_1^{r_1}\dots p_m^{r_m}. \]
    Ova dekompozicija je jedinstvena do na permutaciju polinoma $p_i^{r_i}$.
\end{theorem}

\begin{theorem}
    Ireducibilni polinomi nad $\mathbb{C}$ su oblika:
    \[ x - \alpha, \;\; \alpha \in \mathbb{R}. \]
\end{theorem}

\begin{theorem}
    Ireducibilni polinomi nad $\mathbb{R}$ su oblika:
    \begin{enumerate}
        \item $ x - \alpha, \;\; \alpha \in \mathbb{R}, $
        \item $x^2 + px + q, \;\; p, q \in \mathbb{R}$ td. $D = p^2 - 4q < 0$.
    \end{enumerate}
\end{theorem}

\begin{theorem}
    \textbf{(Ajzenštajnov kriterijum):} Neka je $f(x) = \sum_{i=0}^{n}a_ix^i \in \mathbb{Z}[x]$ i neka postoji prost broj $p$ td. važi:
    \begin{enumerate}
        \item $p\;|\; a_i, \; i \in \{0, 1, \dots, n-1\}$
        \item $p \;\not | \; a_n$
        \item $p^2 \;\not | \; a_0$.
    \end{enumerate}
    Tada je $f(x)$ ireducibilan nad $\mathbb{Q}$.
\end{theorem}

\begin{theorem}
    \textbf{(O racionalnim korenima):}
    Za $P(x) = a_nx^n + \dots + a_0, \; a_i \in \mathbb{Z}$ važi:
\begin{itemize}
    \item $P(\alpha) = 0 \;\land\; \alpha \in \mathbb{Z} \implies \alpha \;| \;a_0$
    \item $\frac{p}{q} \in \mathbb{Q} \;\land\; \operatorname{NZD}(p,q) = 1 \;\land\; P(\frac{p}{q}) = 0 \implies p \;|\; a_0 \;\land\; q \;|\; a_n$
\end{itemize}
\end{theorem}

\subsection{Šetnje, staze i putevi}
\begin{itemize}
    \item \textbf{Šetnja} - niz čvorova i grana $(v_1, e_1, v_2, e_2, \ldots, e_{n-1}, v_n)$ grafa $G$ u kojem se svaka grana pojavljuje između dva sa njom incidentna čvora. \textit{Dužina šetnje} je broj grana koje se pojavljuju u njoj.     \item \textbf{Staza} - šetnja u kojoj se svaka grana pojavljuje najviše jednom. 
    \item \textbf{Put} - šetnja u kojoj se svaki čvor (pa i svaka grana) pojavljuje najviše jednom.
    \item Dva čvora $u$ i $v$ su \textit{povezana} akko postoji šetnja koja počinje u $u$ i završava u $v$. Graf je \textit{povezan} akko su svi njegovi čvorovi međusobno povezani.
    \item Povezanost čvorova u grafu je relacija ekvivalencije. \textit{Komponente povezanosti} grafa su klase ekvivalencije te relacije.
    \item \textbf{Vezivni čvor} - čvor čije uklanjanje povećava broj komponenti povezanosti grafa.
    \item \textbf{Most} (ili vezivna grana) - grana čije uklanjanje povećava broj komponenti povezanosti grafa (za 1).
    \item \textbf{Rastojanje} $\operatorname{dist}(u, v)$ između dva čvora $u$ i $v$ - dužina najkraćeg puta koji ih povezuje. Ako ne postoji takva šetnja, rastojanje je beskonačno.
    \item \textbf{Ekscentricitet} čvora $u$ - $\operatorname{ecc}(u) = \max \{ \operatorname{dist}(u, v)$ | $v \in V \}$.
    \item \textbf{Radijus} grafa $G$ - $\operatorname{rad}(G) = \min \{ \operatorname{ecc}(u)$ | $u \in V \}$.

\begin{theorem}
    Sledeće tvrdnje su ekvivalentne:
    \begin{enumerate}
        \item $T$ je stablo.
        \item $T$ je povezan graf kod kojeg je broj grana za 1 manji od broja čvorova, tj važi $m = n - 1$.
        \item $T$ je maksimalni graf bez ciklusa, tj. dodavanje bilo koje grane u $T$ stvara ciklus.
        \item $T$ je minimalni povezan graf, tj. uklanjanje bilo koje grane iz $T$ ga čini nepovezanim.
        \item Za svaka dva čvora grafa $T$ postoji tačno jedan put koji ih povezuje.
    \end{enumerate}
\end{theorem}
\subsection{Korenska i razapinjuća stabla}
\begin{itemize}
    \item \textbf{Korensko stablo} - stablo u čiji je jedan čvor izdvojen od ostalih. Taj čvor se naziva \textit{koren}.
    \item Ako je $r$ koren stabla i ako važi $\operatorname{dist}(u, r) = i$, kažemo da čvor $u$ pripada \textit{$i$-tom nivou} stabla. Ako $u$ pripada $i$-tom nivou, a $v$ $(i + 1)$-om nivou, i ako su susedni, kažemo da je $u$ \textit{prethodnik} čvora $v$, a $v$ \textit{sledbenik} čvora $u$. Čvor koji nema naslednika nazivamo \textit{list}. Čvor koji nije ni koren ni list nazivamo \textit{unutrašnji čvor}.
    \item \textbf{Korensko $k$-arno stablo} - korensko stablo kod kojeg svaki čvor ima najviše $k$ sledbenika.
    \item \textbf{Potpuno korensko $k$-arno stablo} - korensko $k$-arno stablo kod kojeg svaki čvor koji nije list ima tačno $k$ sledbenika.
    \item \textbf{Razapinjući graf} grafa $G$ - graf koji ima isti skup čvorova kao i $G$ i čiji je skup grana podskup skupa grana grafa $G$. 
    \item \textbf{Razapinjuće stablo} povezanog grafa $G$ - razapinjući graf koji je stablo.
    \item \textbf{Razapinjuća šuma} grafa $G$ - razapinjući podgraf koji se sastoji od razapinjućih stabala komponenata povezanosti grafa $G$.
    \item Ako je $T$ razapinjuće stablo povezanog grafa $G$, onda se dodavanjem proizvoljne grane stablu $T$, pri čemu ona pripada grafu $G$, dobija tačno jedan ciklus u novonastalom grafu. Taj ciklus se naziva \textbf{elementarni ciklus} grafa $G$ u odnosu na razapinjuće stablo $T$.
    \item Broj elementarnih ciklusa povezanog grafa ne zavisi od izbora razapinjućeg stabla. Broj grana svakog razapinjućeg stabla jednak je $n - 1$, a po definiciji svaka grana koja pripada grafu, a ne pripada fiksiranom razapinjućem stablu, određuje jedan elementarni ciklus. Dakle, broj elementarnih ciklusa povezanog grafa $G$ jednak je $m - n + 1$.
\end{itemize}

\begin{theorem}
    Broj elementarnih ciklusa grafa jednak je $m - n + p$, gde je $m$ broj grana grafa, $n$ je broj čvorova grafa, a $p$ je broj komponenti povezanosti grafa.
\end{theorem}
\begin{itemize}
    \item \textbf{Ciklomatički broj} $v$ grafa - broj elementarnih ciklusa grafa.
\end{itemize}

\subsection{Pretrage grafova}
Pretraga grafa je postupak sistematskog obilaska svakog čvora grafa tačno jednom. Rezultat pretrage je \textbf{stablo pretrage} (razapinjuće usmereno stablo).

\begin{itemize}
    \item \textbf{Pretraga po dubini} (DFS - Depth First Search) - „najdublji” mogući put: iz trenutnog čvora se ide ka neobrađenom susedu dokle god je moguće, a zatim se vrši povratak (bektrek) na prethodni čvor. 
    \item \textbf{Pretraga po širini} (BFS - Breadth First Search) - „nivo po nivo”: prvo se obilaze svi neobrađeni susedi početnog čvora, zatim svi neobrađeni susedi tih suseda, i tako redom.
\end{itemize}



\section{Označena stabla}
\subsection{Prebrojavanje označenih stabala}
\begin{theorem}[Kejlijeva teorema]
    Broj označenih stabala sa $n$ čvorova jednak je $n^{n-2}$.
\end{theorem}

\begin{itemize}
    \item \textbf{Kontrakcija} $G \cdot e$ - dobija se od multigrafa $G-e$ identifikovanjem (spajanjem) krajnjih čvorova grane $e$ u jedan čvor.
    \item \textbf{Rekurentna formula} za broj razapinjućih stabala $t(G)$:
    $$t(G) = t(G - e) + t(G \cdot e)$$
\end{itemize}

\subsection{Kodiranje i dekodiranje označenih stabala}
Svako označeno stablo sa $n$ ($n \geq 2$) čvorova može se jednoznačno predstaviti \textbf{Priferovim nizom} dužine $n-2$.
\vfill
\begin{algorithm}[!h]
    \caption{Priferovo kodiranje}
    \begin{enumerate}
        \item \textbf{INPUT}: Stablo $T$ sa $n$ čvorova.
        \item \textbf{WHILE} broj čvorova $> 2$:
        \begin{itemize}
            \item $u$ = list sa najmanjom oznakom;
            \item $v$ = sused čvora $u$;
            \item $s := s + v$; // dodaj $v$ u niz
            \item $T := T - u$; // ukloni $u$ iz stabla
        \end{itemize}
        \item \textbf{RETURN} $s$
    \end{enumerate}
\end{algorithm}
\vfill
\newpage
\begin{algorithm}[!h]
    \caption{Priferovo dekodiranje}
    \begin{enumerate}
        \item \textbf{INPUT}: Priferov niz $s = (s_1, s_2, \ldots, s_{n-2})$.
        \item \textbf{Inicijalizacija}:
        \begin{itemize}
            \item $T = (V, \emptyset)$ gde je $V = \{1, 2, \ldots, n\}$.
            \item $l = (1, 2, \ldots, n)$ // lista čvorova koji mogu biti listovi.
        \end{itemize}
        \item \textbf{WHILE} $s \neq ()$:
        \begin{itemize}
            \item $v$ = najmanji čvor iz $l$ koji se ne nalazi u $s$;
            \item $w = s_1$;
            \item $T := T + \{v, w\}$ // dodaj granu između $v$ i $w$;
            \item $l := l \setminus \{v\}$;
            \item $s := s \setminus \{s_1\}$;
        \end{itemize}
        \item \textbf{Završni korak}: Poveži granom poslednja dva čvora preostala u listi $l$.
        \item \textbf{RETURN} $T$
    \end{enumerate}
\end{algorithm}

\section{Minimalna razapinjuća stabla}

\subsection{Težinski grafovi}
Kod težinskih grafova, svakoj grani $e$ dodeljena je vrednost (težina) $w(e)$, koja predstavlja trošak, udaljenost ili kapacitet veze.

\begin{itemize}
    \item \textbf{Težina grafa} $w(G)$ - suma težina svih grana u grafu:
    $$w(G) = \sum_{e \in E(G)} w(e)$$
    \item \textbf{Minimalno razapinjuće stablo} (MST) - razapinjuće stablo $T$ grafa $G$ za koje je suma težina grana $w(T)$ minimalna.
\end{itemize}

\subsection{Kruskalov algoritam}
Strategija Kruskalovog algoritma je „najpre grana najmanje težine”. Reč je o \textbf{pohlepnom algoritmu} koji u svakom koraku bira lokalno optimalno rešenje.
\newpage
\begin{algorithm}[ht]
    \caption{Kruskalov algoritam}
    \begin{enumerate}
        \item \textbf{INPUT}: Povezan težinski graf $G = (V, E)$.
        \item $T = (V, \emptyset)$ // inicijalizacija praznog stabla.
        \item Sortiraj sve grane $e \in E$ u neopadajući niz po težinama.
        \item \textbf{FOR} svaku granu $e = (u, v)$ iz sortiranog niza:
        \begin{itemize}
            \item \textbf{IF} $T + e$ ne sadrži ciklus:
            \item $T := T + e$ // dodaj granu u stablo.
        \end{itemize}
        \item \textbf{RETURN} $T$
    \end{enumerate}
\end{algorithm}

\begin{theorem}[3.4.1]
    Kruskalov algoritam rezultuje minimalnim razapinjućim stablom (MST) povezanog težinskog grafa.
\end{theorem}

\begin{itemize}
    \item \textbf{Napomena:} Algoritam se može optimizovati uvođenjem brojača dodatih grana. Kada broj grana dostigne $n-1$, algoritam se može završiti, jer je stablo kompletirano.
\end{itemize}

\subsection{Primov algoritam}
Strategija Primovog algoritma zasniva se na principu „najpre najbliži sused”. Za razliku od Kruskalovog algoritma koji grane bira globalno, Primov algoritam gradi stablo postepeno, proširujući trenutni podgraf.

\begin{theorem}[3.4.3]
    Neka je $G = (V, E)$ povezan težinski graf, $U \subset V$ i $e$ grana minimalne težine koja spaja čvor iz skupa $U$ sa čvorom iz skupa $V \setminus U$. Tada postoji minimalno razapinjuće stablo grafa $G$ koje sadrži granu $e$.
\end{theorem}

\begin{algorithm}[ht]
    \caption{Primov algoritam}
    \begin{enumerate}
        \item \textbf{INPUT}: Težinski graf $G = (V, E)$.
        \item $T = (\emptyset, \emptyset)$ // inicijalizacija praznog stabla.
        \item Izaberi proizvoljan čvor $v \in V$ i dodaj ga u $T$.
        \item \textbf{WHILE} broj čvorova u $T < |V|$:
        \begin{itemize}
            \item Nađi granu $e = (u, w)$ minimalne težine takvu da $u \in T$, a $w \notin T$.
            \item Dodaj granu $e$ i čvor $w$ u $T$.
        \end{itemize}
        \item \textbf{RETURN} $T$
    \end{enumerate}
\end{algorithm}

\section{Ojlerovi multigrafovi}
Multigraf je \textbf{Ojlerov} ako sadrži zatvorenu stazu koja svaku granu obilazi tačno jednom. \textbf{Poluojlerov} graf sadrži stazu (ne nužno zatvorenu) koja svaku granu obilazi tačno jednom.

\begin{theorem}[Ojlerova teorema]
    Netrivijalan povezan multigraf je Ojlerov akko svi njegovi čvorovi imaju paran stepen.
\end{theorem}

\begin{theorem}
    Netrivijalan povezan multigraf je poluojlerov akko sadrži tačno 0 ili 2 čvora neparnog stepena.
\end{theorem}


Flerijev algoritam se koristi za pronalaženje Ojlerove staze u grafu. Ključno pravilo pri izboru grane je izbegavanje „mostova” osim ako ne postoji druga mogućnost.

\begin{algorithm}[ht]
    \caption{Flerijev algoritam}
    \begin{enumerate}
        \item \textbf{INPUT}: Ojlerov multigraf $G$.
        \item Izaberi proizvoljan čvor $v_0$. Neka je $s = (v_0)$ trenutna staza.
        \item $H = G$ // radna kopija grafa.
        \item \textbf{WHILE} broj grana u $H > 0$:
        \begin{itemize}
            \item Izaberi granu $e = (u, v)$ iz $H$ takvu da:
            \begin{enumerate}
                \item $u$ je poslednji čvor staze $s$.
                \item $e$ nije most u $H$ (osim ako ne postoji drugi izbor).
            \end{enumerate}
            \item $s := s + e + v$; 
            \item $H := H - e$; // ukloni granu iz pomoćnog grafa.
        \end{itemize}
        \item \textbf{RETURN} $s$
    \end{enumerate}
\end{algorithm}



U slučaju kada graf nije poluojlerov, problem se može modifikovati dodavanjem novih grana ili dozvoljavanjem višestrukih prolazaka kroz iste grane.


\subsection{Problem kineskog poštara}
Problem kineskog poštara (engl. \textit{Route Inspection Problem}) odnosi se na pronalaženje zatvorene šetnje u povezanom težinskom multigrafu koja sadrži sve grane grafa i ima minimalnu ukupnu težinu.

\begin{itemize}
    \item \textbf{Cilj:} Optimizovati rutu tako da se obiđu sve ulice (grane) uz minimalan utrošak vremena (dužine puta).
    \item \textbf{Ojlerov slučaj:} Ako je multigraf Ojlerov, svaka Ojlerova staza predstavlja optimalno rešenje jer se svakom granom prolazi tačno jednom.
    \item \textbf{Opšti slučaj:} Ako graf nije Ojlerov (postoje čvorovi neparnog stepena), neke grane se moraju obići više puta. Problem se tada svodi na „uparivanje” čvorova neparnog stepena i dodavanje puteva minimalne težine između njih kako bi svi čvorovi postali parnog stepena.
\end{itemize}



\begin{algorithm}[ht]
    \caption{Rešavanje problema kineskog poštara (za neojlerov graf)}
    \begin{enumerate}
        \item Identifikuj sve čvorove neparnog stepena u multigrafu.
        \item Odredi sve moguće puteve između parova čvorova neparnog stepena i pronađi onaj skup uparivanja koji daje minimalnu ukupnu težinu.
        \item Dodaj „virtuelne” grane (duplikate postojećih grana) duž najkraćih puteva između odabranih parova, čineći graf Ojlerovim.
        \item Primenom Flerijevog algoritma na tako modifikovan graf pronađi Ojlerovu stazu.
    \end{enumerate}
\end{algorithm}

\section{Hamiltonovi grafovi}
Graf je \textbf{Hamiltonov} ako sadrži ciklus koji prolazi kroz svaki čvor grafa tačno jednom (\textbf{Hamiltonov ciklus}). Graf je \textbf{polu-Hamiltonov} ako sadrži put koji prolazi kroz svaki čvor grafa tačno jednom (\textbf{Hamiltonov put}).

\begin{theorem}[Dirakova teorema]
    Ako graf $G$ ima $n \geq 3$ čvorova i minimalni stepen čvora $\delta(G) \geq \frac{n}{2}$, tada je $G$ Hamiltonov.
\end{theorem}



\subsection*{Problem konjičkog skoka}
Problem konjičkog skoka zahteva obilazak svakog polja šahovske table (ili table formata $m \times n$) skakačem, tako da se na svakom polju stane tačno jedanput.

\begin{itemize}
    \item \textbf{Grafovska interpretacija:} Čvorovi grafa su polja table, a grane predstavljaju dozvoljene poteze skakača.
    \item \textbf{Veza sa teorijom:} Postojanje rešenja ekvivalentno je postojanju Hamiltonovog puta (ili ciklusa) u tako konstruisanom grafu.
    \item \textbf{Napomena:} Za table formata $m \times n$ (gde je $m, n \geq 3$), rešenje postoji osim za tabele formata $3 \times 3$, $3 \times 5$, $3 \times 6$ i $4 \times 4$.
    \item \textbf{Metoda:} Problemi većih dimenzija (npr. $8 \times 8$) često se rešavaju metodom „podela i vladaj” — deljenjem table na manje segmente koji se pojedinačno rešavaju i povezuju.
\end{itemize}



Za razliku od Ojlerovih grafova, ne postoji jednostavna i efikasna teorema (nužan i dovoljan uslov) za proveru postojanja Hamiltonovog ciklusa u opštem grafu, zbog čega je ovaj problem računski složen (NP-kompletan).

\subsection{Problem trgovačkog putnika}
Problem trgovačkog putnika (engl. \textit{Traveling Salesperson Problem - TSP}) zahteva pronalaženje Hamiltonovog ciklusa minimalne težine u potpunom težinskom grafu.

\begin{itemize}
    \item \textbf{Cilj:} Obići svaki grad (čvor) tačno jednom i vratiti se u početni, uz minimalnu ukupnu dužinu puta.
    \item \textbf{Kompleksnost:} Problem je \textbf{NP-težak}, što znači da ne postoji poznat algoritam koji bi ga rešio u polinomijalnom vremenu za sve ulaze.
    \item \textbf{Pristupi:}
    \begin{itemize}
        \item \textit{Egzaktni algoritmi:} Pretraga svih mogućih Hamiltonovih ciklusa (neefikasno za veliki broj gradova).
        \item \textit{Heuristički algoritmi:} Traže \textbf{suboptimalna} rešenja (rešenja "bliska" optimalnom) u razumnom vremenu.
    \end{itemize}
\end{itemize}



\subsection*{3-heuristika}
Ova heuristika poboljšava početni Hamiltonov ciklus kroz iterativne promene (lokalna pretraga).

\begin{algorithm}[ht]
    \caption{3-heuristika za TSP}
    \begin{enumerate}
        \item \textbf{INPUT}: Hamiltonov ciklus $s_1$ potpunog težinskog grafa.
        \item \textbf{WHILE} kriterijum zaustavljanja nije ispunjen:
        \begin{itemize}
            \item Ukloni 3 nesusedne grane iz trenutnog ciklusa $s_i$, čime se dobijaju 3 zasebna puta.
            \item Ponovo poveži krajeve tih puteva na sve moguće načine da se formira novi Hamiltonov ciklus.
            \item Izaberi ciklus $s_{i+1}$ sa minimalnom težinom među svim novonastalim ciklusima.
            \item \textbf{IF} $w(s_{i+1}) < w(s_i)$, prihvati $s_{i+1}$ kao trenutni ciklus.
        \end{itemize}
        \item \textbf{RETURN} $s_i$ (suboptimalni ciklus)
    \end{enumerate}
\end{algorithm}

\begin{itemize}
    \item \textbf{Napomena:} Strategija „uvek idi do najbližeg suseda” (Greedy) je najjednostavnija heuristika, ali često ne daje optimalno rešenje. 3-heuristika daje znatno bolje rezultate jer omogućava „preslaganje” ciklusa.
\end{itemize}

\section{Planarni grafovi}
Graf je \textbf{planaran} ako se može nacrtati u ravni tako da se njegove grane nigde ne seku (osim u čvorovima). Svaki takav crtež deli ravan na jednu ili više \textbf{oblasti} (jedna od njih je uvek beskonačna).



\begin{theorem}[Ojlerova teorema za planarne grafove]
    Za svaki povezan planaran graf $G$ važi:
    $$n - m + f = 2$$
    gde su $n$ broj čvorova, $m$ broj grana, a $f$ broj oblasti na koje graf deli ravan.
\end{theorem}

\textbf{Posledice Ojlerove teoreme:}
\begin{itemize}
    \item Za svaki prost planaran graf sa $n \geq 3$ važi:
    \[ m \le 3n - 6 \] 
    \item Ako prost planaran graf ne sadrži cikluse dužine 3 (npr. bipartitan je), onda važi stroža nejednakost:
    \[ m \le 2n - 4 \]
\end{itemize}

\subsection*{Kuratovskijeva teorema}
Da bismo definisali karakterizaciju planarnih grafova, koristimo pojam \textbf{potpodele} (subdivision), koja nastaje umetanjem novih čvorova na postojeće grane.



\begin{theorem}[Kuratovskijeva teorema]
    Graf je planaran ako i samo ako ne sadrži podgraf koji je izomorfan nekoj potpodi grafa $K_5$ (kompletan graf sa 5 čvorova) ili $K_{3,3}$ (kompletan bipartitan graf sa 3+3 čvora).
\end{theorem}

\begin{itemize}
    \item \textbf{Značaj:} Ovo je fundamentalni rezultat koji nam omogućava da testiramo planarnost bilo kog grafa identifikacijom tzv. "zabranjenih" struktura (Kuratovskijevi grafovi).
\end{itemize}

\section{Sparivanja u grafovima}
\textbf{Sparivanje} (matching) u grafu $G$ je podskup skupa grana takav da nijedne dve grane nemaju zajednički čvor (nisu susedne).

\begin{itemize}
    \item \textbf{Maksimalno sparivanje} - sparivanje kojem se ne može dodati nova grana, a da ostane sparivanje.
    \item \textbf{Najveće sparivanje} - sparivanje sa najvećim mogućim brojem grana.
    \item \textbf{Savršeno sparivanje} - sparivanje u kojem je svaki čvor grafa incidentan sa tačno jednom granom sparivanja.
\end{itemize}



\subsection*{Beržova teorema i proširenje}
\begin{itemize}
    \item \textbf{M-alternirajući put} - put kod kojeg grane naizmenično pripadaju i ne pripadaju sparivanju $M$.
    \item \textbf{M-uvećani put} - M-alternirajući put čiji su krajnji čvorovi slobodni (nisu pokriveni sparivanjem $M$).
\end{itemize}

\begin{theorem}[Beržova teorema]
    Sparivanje $M$ u grafu $G$ je najveće ako i samo ako $G$ ne sadrži M-uvećani put.
\end{theorem}

\subsection*{Problem rasporeda poslova (Asignacija)}
Problem se modeluje pomoću bipartitnog grafa gde jedna particija čvorova predstavlja radnike, a druga poslove.

\begin{itemize}
    \item \textbf{Jednostavan slučaj:} Ako su svi radnici jednako efikasni, problem se svodi na pronalaženje \textbf{najvećeg sparivanja} u bipartitnom grafu. Za to se koristi \textbf{Mađarski algoritam}.
    \item \textbf{Složeniji slučaj:} Ako radnici imaju različitu efikasnost, koriste se težinski bipartitni grafovi gde težina grane predstavlja cenu ili vreme obavljanja posla. Ovde se primenjuje \textbf{Kun-Mankresov algoritam}.
\end{itemize}

\section{Bojenje grafova}
Bojenje grafova podrazumeva dodeljivanje boja elementima grafa (čvorovima ili granama) uz određene uslove.

\begin{itemize}
    \item \textbf{Pravilno bojenje čvorova} - funkcija kojom se susednim čvorovima dodeljuju različite boje.
    \item \textbf{Hromatski broj $\chi(G)$} - minimalan broj boja potrebnih za pravilno bojenje čvorova. To je invarijanta grafa.
    \item \textbf{Linijski graf $L(G)$} - graf čiji čvorovi odgovaraju granama grafa $G$, a susedni su ako su odgovarajuće grane u $G$ susedne. Bojenje grana grafa $G$ ekvivalentno je bojenju čvorova grafa $L(G)$.
\end{itemize}

\subsubsection{Teoreme i svojstva}
\begin{itemize}
    \item \textbf{Kompletni grafovi:} $\chi(K_n) = n$.
    \item \textbf{Bipartitni grafovi:} Za graf sa bar jednom granom, $\chi(G) = 2$ (bihromatski grafovi).
    \item \textbf{Ciklus $C_n$:} $\chi(C_n) = 2$ za $n=2k$, odnosno $\chi(C_n) = 3$ za $n=2k-1$.
    \item \textbf{Gornja granica:} $\chi(G) \leq \Delta + 1$.
    \item \textbf{Bruks teorema:} Jednakost $\chi(G) = \Delta + 1$ važi akko je graf kompletan ili ciklus neparne dužine.
\end{itemize}

\subsection{Bojenje mapa}
Problem bojenja mapa datira iz XIX veka i odnosi se na bojenje teritorija tako da susedne teritorije (one koje dele granicu dužine veće od nule) imaju različite boje.

\begin{itemize}
    \item \textbf{Modelovanje:} Svaka mapa se može predstaviti kao planaran graf gde čvorovi odgovaraju teritorijama, a grane granicama. Problem se svodi na određivanje hromatskog broja tog planarnog grafa.
    \item \textbf{Teorema o četiri boje:} Svaki planaran graf je 4-obojiv ($\chi(G) \leq 4$).
    \item \textbf{Istorijat:} Hipotezu je 1890. godine podržao Hejvud (dokazao 5-obojivost), dok su Epl i Haken tek 1976. godine dokazali teoremu o 4 boje uz pomoć računara.
\end{itemize}


\subsection*{Primene i algoritmi}
\begin{itemize}
    \item \textbf{Pohlepna heuristika:} Čvorovi se boje redom ($v_1, \dots, v_n$), a svakom se dodeljuje prva boja koja nije iskorišćena za njegove susede. Iako efikasan, ovaj algoritam ne garantuje minimalan broj boja ($\chi$).
    \item \textbf{Raspored ispita:} Ispiti su čvorovi, a grane povezuju one koje student polaže istovremeno. Potreban broj termina = $\chi(G)$.
    \item \textbf{Skladištenje materijala:} Materije su čvorovi, a grane povezuju nekompatibilne supstance. Minimalan broj skladišnih prostorija = $\chi(G)$.
\end{itemize}

\section{Rastojanja u usmerenim težinskim grafovima}
\subsection{Dajkstrin algoritam}
\begin{algorithm}[ht]
    \caption{Dajkstrin algoritam}
    \begin{enumerate}
        \item \textbf{INPUT}: Usmeren težinski graf $G$ i početni čvor $r$.
        \item \textbf{Inicijalizacija}:
        \begin{itemize}
            \item $prethodnik(u) = undefined$ za sve $u \in V$.
            \item $rastojanje(r) = 0$, a $rastojanje(u) = \infty$ za sve $u \in V \setminus \{r\}$.
            \item $S = \emptyset$ (skup razmotrenih čvorova).
        \end{itemize}
        \item \textbf{WHILE} $S \neq V$:
        \begin{itemize}
            \item Nađi $u \in V \setminus S$ takav da je $rastojanje(u)$ minimalno.
            \item $S := S \cup \{u\}$.
            \item \textbf{FOR} svaka grana $(u, v)$ gde $v \in V \setminus S$:
            \begin{itemize}
                \item \textbf{IF} $rastojanje(u) + w(u, v) < rastojanje(v)$:
                \begin{itemize}
                    \item $prethodnik(v) = u$
                    \item $rastojanje(v) = rastojanje(u) + w(u, v)$
                \end{itemize}
            \end{itemize}
        \end{itemize}
        \item \textbf{Rekonstrukcija puteva} (za svaki $u \in S$):
        \begin{itemize}
            \item $v = u$, $put = (u)$.
            \item \textbf{WHILE} $prethodnik(v) \neq undefined$:
            \begin{itemize}
                \item $v = prethodnik(v)$, $put = v + put$.
            \end{itemize}
        \end{itemize}
        \item \textbf{RETURN} $put, rastojanje(u)$
    \end{enumerate}
\end{algorithm}

\begin{itemize}
    \item \textbf{Napomena:} U prvom delu, algoritam određuje prethodnika svakog čvora i izračunava najkraća rastojanja. Drugi deo algoritma služi za rekonstrukciju samih putanja koristeći informacije o prethodnicima. Moguća je modifikacija algoritma kojom se direktno pamte nizovi čvorova koji čine najkraći put.
\end{itemize}

\begin{theorem}[Teorema 3.10.3]
    Dajkstrin algoritam daje najkraće puteve od jednog čvora usmerenog težinskog grafa do svih čvorova tog grafa.
\end{theorem}

\subsubsection*{Linearni funkcional i dualni prostor}
Ako je $V$ vektorski prostor nad poljem $\mathbb{F}$, tada se svako linearno preslikavanje \[f: V \to \mathbb{F}\] zove \textbf{linearni funkcional}. Ako je $\dim V < \infty$, onda je $V^* = \operatorname{Hom} (V, \mathbb{F})$ vektorski prostor i važi $\dim V = \dim V^*$. Dakle, $f(x_1, \dots, x_n) = a_1x_1 + \dots + a_nx_n \in \mathbb{R}$. $V^*$ nazivamo \textbf{dualni prostor} prostora $V$.


Zbog teoreme o rangu i defektu važi \[ \operatorname{r} (f) \leq 1 \;\;\text{i}\;\; \operatorname{d}(f) \geq \dim V - 1,\] tj. ako $f \neq 0$, tada postoji $0 \neq v \in V$ td. $f(v) \neq 0$.
\\
\subsubsection{Dualna baza}
    Neka je $V$ vektorski prostor nad $\mathbb{F}$ ($n = \dim V < \infty$) i $e = (e_1, \dots, e_n)$ neka njegova baza.
     Postoji jedinstvena baza $e^* = (e_1^*, \dots, e_n^*)$ prostora $V^* = \operatorname{Hom} (V, \mathbb{F})$ takva da je:
    \begin{enumerate}
        \item \[
        e^*_j(e_i) = \delta_{ij}=\begin{cases}
            1, &j=i\\
            0, &j \neq i
        \end{cases}
    \]
    \item svaki linearni funkcional $f: V \to \mathbb{F}$ je oblika\[ f = \sum_{j=1}^{n} f(e_j)e_j^* \] 
    \item za svaki vektor $x \in V$ važi \[ x = \sum_{j=1}^{n} e^*_j(x)e_j \]
    \end{enumerate}
    Baza $e^*$ se zove \textbf{dualna baza} bazi $e$. \\
    
    \noindent\underline{\textit{Intuicija:}} Dualna baza rešava sledeći problem. Recimo da imamo vektor $x \in V$ zadat u standardnim koordinatama i nestandardnu bazu $e$ prostora $V$. Hoćemo da izrazimo $x$ kao linearnu kombinaciju vektora baze $e$. Dualna baza se sastoji od linearnih funkcionala koji rade baš ovo - pretvaraju standardne koordinate u koordinate u nekoj bazi $e$. Funkcionali $e_j^*$ su upravo koeficijenti u redovima inverzne matrice prelaska.
\newpage
\subsubsection*{Primer: Određivanje dualne baze}
Neka je $V = \mathbb{R}^3$ i neka je $V = \mathcal{L}\left(\{e_1 = (2, 1), \;e_2 = (3,1)\}\right)$. Odredimo bazu $e^*$. Znamo da važi $e^*_j(e_i) = \delta_{ij}$. Dakle, 
\[ e_1^*(e_1) = e_1^*((2, 1)) = 2e_1^*((1, 0)) + e_1^*((0, 1)) = 1 \]
\[ e_1^*(e_2) = e_1^*((3, 1)) = 3e_1^*((1, 0)) + e_1^*((0, 1)) = 0 \]
Rešavanjem ovog sistema, dobijamo $e_1^*((1, 0)) = -1$ i $e_1^*((0, 1)) = 3$. Analognim postupkom, za $e_2^*$ dobijamo $e_2^*((1, 0)) = 1$ i $e_2^*((0, 1)) = -2$. Sada, za proizvoljan
vektor $v = (x, y)$ važi 
\[ e_1^*(v) = xe_1^*((1, 0)) + ye_1^*((0, 1)) = -x + 3y \]
\[ e_2^*(v) = xe_2^*((1, 0)) + ye_2^*((0, 1)) = x - 2y \]
Dakle, $V^* = \mathcal{L} \left(\left\{ e_1^* , e_2^* \right\}\right)  = \mathcal{L} \left(\left\{ -x + 3y , x - 2y \right\}\right)$.

\subsubsection*{Primer: Dualna baza i matrica prelaska}
Neka je data nestandardna baza prostora $\mathbb{R}^2$:
\[
e_1 = (1, 1), \quad e_2 = (1, -1).
\]
Kolone matrice prelaska $P$ su vektori baze $e$:
\[
P = \begin{bmatrix} 1 & 1 \\ 1 & -1 \end{bmatrix} \] 
Računamo inverznu matricu:
\[ P^{-1} = \frac{1}{-2} \begin{bmatrix} -1 & -1 \\ -1 & 1 \end{bmatrix}
= \begin{bmatrix} \frac{1}{2} & \frac{1}{2} \\[4pt] \frac{1}{2} & -\frac{1}{2} \end{bmatrix}.
\]
Vrste matrice $P^{-1}$ su koeficijenti funkcionala dualne baze $e^*$:
\[
e_1^*(x,y) = \frac{1}{2}x + \frac{1}{2}y, \quad
e_2^*(x,y) = \frac{1}{2}x - \frac{1}{2}y.
\]
Neka je vektor $v \in \mathbb{R}^2$ zadat u \textit{standardnim koordinatama}:
\[
v = (7, 3).
\]
Koordinate vektora $v$ u bazi $e$ dobijamo množenjem kolone standardnih koordinata matricom $P^{-1}$ (tj. primenom funkcionala dualne baze):
\[
[v]_e = P^{-1} \cdot [v]_{\text{std}}
= \begin{bmatrix} \frac{1}{2} & \frac{1}{2} \\[4pt] \frac{1}{2} & -\frac{1}{2} \end{bmatrix}
\begin{bmatrix} 7 \\ 3 \end{bmatrix} = \begin{bmatrix} 5 \\ 2 \end{bmatrix}.
\]
Dakle, standardni vektor $(7, 3)$ zapisan u bazi $e$ ima koordinate:
\[
\boxed{v = 5e_1 + 2e_2}
\]

\subsubsection{Anihilator}
Svaki linearni funkcional je oblika $f(x) = \sum_{j=1}^{n}x_j\xi_j$, gde je $\xi_j=f(e_j)\in\mathbb{F}$. Jezgro ovog (ne-nula) preslikavanja 
$\operatorname{Ker} f = \{x \in V \;|\; f(x) = 0\}$ je očigledno hiperravan u $V \cong \mathbb{F}^n$, koja prolazi kroz koordinatni početak.
I obrnutno, svaka hiperravan na $V$ je jezgro nekog ne-nula linearnog funkcionala na $V$. 

Neka je $H = \operatorname{Ker} f$ hiperravan u $V$
i neka je $\mathcal{B}_H = \{a_1, a_2, \dots, a_{n-1}\}$ baza za $H$. Dopunimo je vektorom $a_0$ do baze za $V$. Tada je \[H = \{x \in V\;|\; f(x) = 0\},\] gde
je $f$ jedinstveni linearni funkcional koji zadovoljava \[f(a_0) = 1 \footnote{mogli smo odabrati i bilo koju drugu vrednost $\alpha \neq 0$ - jezgro ostaje isto} \;\;\text{i}\;\; 
f(a_1) = f(a_2) = \dots = f(a_{n-1})=0.\]

Sa druge strane, skup $A = \{g \in V^* \;|\; g(a_0) = 0\}$ je \textbf{anihilator} $\{a_0\}^0$ vektora $a_0$. Kako je $\dim A = \dim H = n-1$, $A$ i $H$ su izomorfni. Ovo možemo uopštiti na 
bilo koji podskup $S$ vektorskog prostora $V$ i dobiti \textbf{anihilator} skupa $S:$ \[S^0 = \{ f \in V^* \;|\; f(x) = 0, \forall x \in S \}\] \underline{\textit{Intuicija:}} Geometrijski, ako je $S$ podskup prostora $V$, njegov 
anihilator $S^0$ je skup svih linearnih funkcionala $f(x) = a_1x_1 + a_2x_2 +  \dots +a_{n}x_{n}$ takvih da je vektor $(a_1, \dots, a_{n})$ normalan na svaki vektor $s \in S$. Drugim rečima: $L^0$ - sva linearna pitanja čiji je odgovor 0 na svakom vektoru iz $L$.

\begin{theorem}
    Ako je $V$, $\dim V < \infty$, vektorski prostor nad $F$ i $L$ njegov potprostor, tada važi
    \[ \dim L + \dim L^0 = \dim V \]
\end{theorem}
\noindent\underline{\textit{Intuicija:}} Ako je npr. $V = \mathbb{R}^3$ i $\dim L = 2$, tada se svi vektori normalni na $L$ nalaze na pravoj $\alpha x$, $\alpha \in \mathbb{R}, \; x \in L^\perp$,\; tj. $\dim L^0 = 1$. Zapravo, 
prostori $L^0$ i $L^\perp$ su izomorfni ukoliko postoji skalarni proizvod na $V$. 
\\


\noindent\textbf{Posledica 1.} \textit{
    Neka je $\dim V = n,\; \dim L = l,\; L \leq V$. Tada je00 
    \[ L = H_1 \cap H_2 \cap \dots \cap H_{n-l},  \] gde su $H_i$ hiperravni od $V$.}\\
\noindent\underline{\textit{Intuicija:}} Neka je, na primer, $V = \mathbb{R}^3$. Ako je $\dim L = 2$, tada je $L$ jedna hiperravan (tj. ravan) iz $\mathbb{R}^3$. Sa druge strane, ako je $\dim L = 1$, tada je $L$ prava iz $\mathbb{R}^3$,
tj. presek dve ravni iz $\mathbb{R}^3$. Slično, ako je $L$ tačka iz $\mathbb{R}^3$, tj. $\dim L = 0$, tada je $L$ zapravo presek tri ravni iz $V$.
\\


\noindent\textbf{Posledica 2.}\textit{
    Neka je $\dim V < \infty$ i $L \leq V, \; M \leq V$. Tada je $L = M \iff L^0 = M^0$.}\\
\noindent\underline{\textit{Intuicija:}} Neka je npr. $V = \mathbb{R}^3$ i $L$ i $M$ dve ravni. Znamo da je svaka ravan potpuno određena vektorom normale i jednom tačkom koju sadrži. Kako anihilator upravo zamišljamo
kao prostor čiji su svi vektori normalni na dati skup (u ovom slučaju v. prostor) i kako svaki anihilator zadrži tačku nula, imamo baš ono što nam treba.

\subsubsection{Refleksivnost}
Za v. prostor $V$, $\dim V < \infty$, nad $\mathbb{F}$ posmatramo niz $V, V^*, V^{**}, V^{***}, \dots$. Svi elementi ovog niza su prostori nad istim poljem $\mathbb{F}$ i imaju istu dimenziju $\dim V$. Dakle, 
svi ovi prostori su izomorfni.

Za svako $x \in V$ i $f \in V^*$ možemo definisati linearno preslikavanje $\phi:V \to V^{**}:$ \[\phi(x)[f] \;  = f(x). \;\footnote{Napomena: $\phi: V \to V^{**}$, ali $\phi(x) : V^* \to \mathbb{F}$.} \]
\underline{\textit{Intuicija:}} Zamislimo stvari na sledeći način. 

\begin{itemize}
    \item $V$ sadrži neke objekte $x, y, z, \dots$, čije osobine možemo da merimo. Ove objekte zovemo vektori. Svaka osobina je upravo po jedna komponenta vektora.
    \item $V^*$ su svi mogući eksperimenti (merenja) koje možemo da uradimo na objektima iz $V$. Ove eksperimente zovemo linearnim funkcionalima. Jedan funkcional može biti 
    npr. "Kolika je $x$-koordinata?", drugi "Kolika je $y$-koordinata?", treći "Koliko je $x + y - 2z$?"\ itd. Dakle, $V^*$ nije skup rezultata, već svih pitanja koje možemo
    postaviti vektoru.
    \item Šta sve znamo o konkretnom vektoru, npr. $x = (3, -1, 5)$? Znamo kako odgovara na svako moguće pitanje (tj. koju vrednost daje za svaki mogući eksperiment). Dakle, jedan
    vektor određuje beskonačnu tabelu sa pitanjima i odgovorima:
    \begin{table}[!h]
        \centering
        \begin{tabular}{cc}
            \toprule
            Pitanje & Odgovor \\ \midrule
            $f_1$ & 3 \\
            $f_2$ & 2 \\
            $f_3$ & 42 \\
            \vdots & \vdots \\
            \bottomrule
        \end{tabular}
    \end{table}

    \item Šta radi $\phi(x)$? On pravi ovu tabelu. On kaže "{daj mi pitanje $f$ i daću ti odgovor na $f$ za vektor $x$}". Dakle, $\phi(x): f \mapsto f(x)$. Zašto je ovo zapravo element
    prostora $V^{**}$? Znamo da svaki linearni funkcional slika element prostora u broj (opštije, u element polja $\mathbb{F}$). Pošto je sada $V^*$ prostor, element mora slikati $V^* \to \mathbb{F}$.
    Dakle, uzme funkcional i vrati broj. Koji broj? Baš njegov rezultat na $x$.
    \item Ovim pristupom vektor više nije samo strelica. Možemo ga posmatrati kao mašinu koja zna odgovor na svako moguće linearno pitanje. Ovo je potpuno nova filozofija. Evo još jedne jako prirodne analogije.
    Zamislimo osobu. Ne opisujemo je fotografijom. Opisujemo je odgovorom na sva moguća pitanja. Ime? Marko. Visina? 183. Masa? 79. Godine? 25. I tako dalje. Ako znamo odgovore na sva moguća pitanja, onda znamo i osobu.
    Ne treba nam fotografija. Isto tako, ne moramo znati koordinate vektora. Potpuno je određen odgovorom na sva moguća pitanja.
    \item Dakle, ne treba gledati na $V^{**\dots*}$ kao merač merača merača $\dots$ merača, već razmišljati na gore navedeni način.
\end{itemize}

\subsection{Dualno (transponovano) preslikavanje}
Neka su $U$ i $V$ vektorski prostori nad $\mathbb{F}$ i $A \in \operatorname{Hom} \left(U, V\right)$. Tada $A$ prirodno indukuje linearno preslikavanje $A^*: V^* \to U^*$ formulom
\[ (A^*[f])(x) := A^*[f](x) = (f \circ A)(x) = f(A(x)) \]
Linearni operator $A^*$ zove se \textbf{dualni (transponovani, adjugovani)} operator od $A$.
\begin{theorem}
    Neka su $U$ i $V$ vektorski prostori nad $\mathbb{F}$ i $A \in \operatorname{Hom}\left(U, V\right)$. Tada postoji jedinstveni linearni operator $A^* 
    \in \operatorname{Hom}\left(U^*, V^*\right)$ takav da je $A^*[f](x) = f(A(x)), \linebreak \forall f \in V^*, \; \forall x \in U$.
\end{theorem}
\begin{theorem}
    Neka su $U$ i $V$ vektorski prostori nad $\mathbb{F}$, $\operatorname{dim} U, \operatorname{dim} V < \infty$, i \linebreak$A \in \operatorname{Hom}\left(U, V\right)$. Tada
    \begin{enumerate}
        \item $\ker A^* = (\operatorname{Im} A)^0$ \\
        \underline{\textit{Intuicija:}} $\ker A^* = \left\{f \in V^* \; | \; f\left(A\left(x\right)\right) = 0, \; \forall x \in U\right\} = (\operatorname{Im} A )^0$
        \item $\operatorname{Im} A^* = (\ker A)^0$
        \item $\operatorname{r}(A^*) = \operatorname{r}(A)$\\
        \underline{\textit{Intuicija:}} Očigledno iz teoreme \ref{ATjednakoAzvezda} ($\mathcal{A^T} = \mathcal{A}^*$).
    \end{enumerate}
\end{theorem}
\begin{theorem}
    \label{ATjednakoAzvezda}
    Neka su $U$ i $V$ konačnodimenzioni v. prostori nad $\mathbb{F}$ i $A \in \operatorname{Hom}\left(U, V\right)$. Neka je $e$ neka baza od $U$ i $e^*$ njoj dualna 
    baza od $U^*$, i neka je $f$ neka baza od $V$ i $f^*$ njoj dualna baza od $V^*$. Ako sa $\mathcal{A}$ obeležimo matricu operatora $A$ u paru baza $e$ i $f$, a sa
    $\mathcal{A}^*$ matricu operatora $A^*$ u paru baza $f^*$ i $e^*$, tada je \[\mathcal{A}^\mathcal{T} = \mathcal{A}^*.\]
\end{theorem}
\noindent\textbf{Posledica 1.} Neka je $\mathcal{A}$ matrica nad poljem $\mathbb{F}$. Tada se rang od $\mathcal{A}$ po vrstama podudara sa rangom od $\mathcal{A}$ po kolonama.\\

\noindent\underline{\textit{Intuicija:}} Dakle, $A$ slika $U$ u $V$, a $A^*$ slika $V^*$ u $U^*$. Takođe, vidimo da je transponovana matrica preslikavanja $A$ upravo matrica preslikavanja $A^*$. Znamo da ako neko linearno preslikavanje 
$T$ slika $U$ u $V$ i $\operatorname{dim} U = n, \; \operatorname{dim} V = m$, tada će njegova matrica $\mathcal{T}$ biti dimenzija $\operatorname{dim} V \times \operatorname{dim} U = m \times n$,
a $\mathcal{T^T}$dimenzija $n \times m$, tj. oblika
\[
    \mathcal{T} = \begin{bmatrix}
        a_{11} & a_{12} & \dots & a_{1n} \\
        a_{21} & a_{22} & \dots & a_{2n} \\
        \vdots & \vdots & \ddots & \vdots \\
        a_{m1} & a_{m2} & \dots & a_{mn}
    \end{bmatrix} \implies 
    \mathcal{T^T} = \begin{bmatrix}
        a_{11} & a_{21} & \dots & a_{m1} \\
        a_{12} & a_{22} & \dots & a_{m2} \\
        \vdots & \vdots & \ddots & \vdots \\
        a_{1n} & a_{2n} & \dots & a_{nm}
    \end{bmatrix},
\] što znači da intuitivno ima smisla (jer je $V\cong V^*$) da $\mathcal{T}$ odgovara $A$, a $\mathcal{T^T}$ odgovara $A^*$. Iz ovog razloga
važi $(AB)^\mathcal{T} = B^\mathcal{T} A^\mathcal{T}$. Naime, ako važi $U \xrightarrow{AB} W$, tj. $U \xrightarrow{B} V \xrightarrow{A} W$, 
u duhu dosadašnje diskusije dobijamo $W^* \xrightarrow{A^\mathcal{T}} V^* \xrightarrow{B^\mathcal{T}} U^*$. Dakle, ovo je intuicija iza trasponovane matrice. Da rezimiramo:
\[ \boxed{u \in U \xrightarrow{A} A(u) \in V \xrightarrow{f} f(A(u)) \in \mathbb{F}}, \;\; f: V \to \mathbb{F}, \;A: U \to V \]




\subsection{Zavisnost matrice operatora od baze}
Neka su $U, V, W$ konačnodimenzioni vektorski prostori nad poljem $\mathbb{F}$ i neka je 
$\dim U = m$, $\dim V = n$ i $\dim W = l$. 
Neka su dati linearni operatori 
\[
A: U \to V, \qquad B: V \to W,
\]
i neka su $e = (e_1, \dots, e_m)$, $f = (f_1, \dots, f_n)$ i $g = (g_1, \dots, g_l)$ redom baze prostora $U, V, W$.

\medskip
\noindent
\subsubsection*{Matrica operatora u datim bazama} 
Matrica operatora $A$ u paru baza $(e, f)$ definisana je sa:
\[
Ae_j = \sum_{i=1}^{n} \alpha_{ij} f_i, \qquad j = 1, \dots, m,
\]
gde je $\mathcal{A} = (\alpha_{ij}) \in \mathbb{M}_{n \times m}(\mathbb{F})$.
Slično, matrica operatora $B$ u paru baza $(f, g)$ je:
\[
Bf_k = \sum_{r=1}^{l} \beta_{rk} g_r, \qquad k = 1, \dots, n.
\]

Ako je $C = B \circ A : U \to W$, tada je matrica kompozicije u paru baza $(e, g)$ data kao proizvod matrica (obrnutim redom):
\[
\boxed{\mathcal{C} = \mathcal{B} \cdot \mathcal{A}}
\]
gde je $\mathcal{C} \in \mathbb{M}_{l \times m}(\mathbb{F})$, $\mathcal{B} \in \mathbb{M}_{l \times n}(\mathbb{F})$, a $\mathcal{A} \in \mathbb{M}_{n \times m}(\mathbb{F})$.

\medskip
\noindent
\subsubsection*{Matrica prelaska} 
Neka je $T_{ee'}$ matrica prelaska sa baze $e$ na bazu $e'$. Tada za koordinate proizvoljnog vektora $x \in U$ važi:
\[
x(e) = T_{ee'} \, x(e'),
\]
gde su $x(e)$ i $x(e')$ koordinate vektora $x$ u tim bazama.

\medskip
\noindent
\subsubsection*{Promena baze linearnog operatora}
Neka je $A: U \to V$ linearni operator. Ako su $e, e'$ baze od $U$ i $f, f'$ baze od $V$, tada je veza između matrica operatora $A$ u parovima baza $(e, f)$ i $(e', f')$ data sa:
\[
\boxed{\mathcal{A}(f', e') = T_{ff'}^{-1} \, \mathcal{A}(f, e) \, T_{ee'}}
\]
gde su $T_{ee'}$ i $T_{ff'}$ odgovarajuće matrice prelaska.

Specijalno, ako je $V = U$ i biramo istu bazu za polazni i dolazni prostor ($f = e$, $f' = e'$), formula se svodi na relaciju \textbf{sličnosti}:
\[
\boxed{\mathcal{A}(e') = T_{ee'}^{-1} \, \mathcal{A}(e) \, T_{ee'}}
\]
Tada kažemo da su matrice $\mathcal{A}(e)$ i $\mathcal{A}(e')$ \textbf{slične}.

\subsection{Simetrična grupa $\mathbb{S}_n$}
\begin{itemize}
    \item $\mathbb{S}_n = \left\{1, 2, \dots, n\right\}, \;\forall n \in \mathbb{N}$
    \item Bijekcije skupa $\mathbb{S}_n$ nazivamo \textbf{permutacije}, a grupu $(\mathbb{S}_n, \circ)$ \textbf{grupa permutacija} ili \textbf{simetrična grupa}.
    \item Zapis: \[
\sigma = \begin{pmatrix} 
1 & 2 & \cdots & n \\ 
\sigma(1) & \sigma(2) & \cdots & \sigma(n) 
\end{pmatrix} \in \mathbb{S}_n
\]
\end{itemize}

\subsection{Parnost permutacija}
\begin{itemize}
    \item \textbf{Inverzija permutacije $\sigma$} - uređeni par $(i, j)$ td. $i < j$ i $\sigma(i) > \sigma(j)$
    \item $I(\sigma)$ - Broj inverzija perm. $\sigma$
    \item $\sigma$ je parna $\iff I(\sigma)$ parno
    \item Preslikavanje za određivanje parnosti: \[ \operatorname{sgn}(\sigma) = \prod_{i < k < n} \frac{\sigma(k) - \sigma(i)}{k - i} \in \left\{-1, 1\right\} \]
    \item $A_n = \ker \operatorname{sgn} = \left\{\sigma \in \mathbb{S}_n \; | \; \operatorname{sgn}(\sigma) = 1\right\}$ - \textbf{alternirajuća podgrupa od $\mathbb{S}_n$ (podgrupa parnih permutacija)} 
\end{itemize}

\begin{theorem}\
    \begin{enumerate}
        \item $\operatorname{sgn}: (\mathbb{S}_n, \circ) \to (\left\{-1, 1\right\}, \cdot)$ je homomorfizam grupa, tj. $ \operatorname{sgn}(\sigma \cdot \tau) = \operatorname{sgn}(\sigma) \cdot \operatorname{sgn}(\tau) $
        \item $\ker \operatorname{sgn} = A_n$ je grupa.
        \item $||A_n|| = \frac{n!}{2}$
    \end{enumerate}
\end{theorem}

\subsection{Ciklus}
\begin{itemize}
    \item Permutacija $\sigma$ je \textbf{ciklus dužine $k$} ako postoji $(i_1, \dots, i_k) \in \mathbb{S}_n^k$ td. \[  \sigma(i_1) = i_2, \sigma(i_2)
    = i_3, \dots, \sigma(i_k) = i_1 \land (\forall j \in \mathbb{S}_n \setminus \{i_1, \dots, i_k\})\;\sigma(j) = j \]
    \item Ciklusi dužine 2 se zovu \textbf{transpozicije}. Transpozicije su neparne.
    \item Disjunktni ciklusi komutiraju. (Očigledno, jer, na primer, ciklus $(124)$ "ne pipa" broj 3.)
    \item \textbf{Red permutacije:} $|\sigma| = \min\{j \in \mathbb{N} \; | \; \sigma^j = \operatorname{id}\}$
\end{itemize}
\newpage
\begin{theorem}
    Svaka permutacija $\sigma \in \mathbb{S}_n$ se može predstaviti kao proizvod
    \begin{enumerate}
        \item disjunktnih ciklusa \\
        \underline{primer:} $\begin{pmatrix} 
1 & 2 & 3 & 4 & 5 & 6 & 7 & 8 \\ 
3 & 8 & 6 & 5 & 2 & 1 & 4 & 7 
\end{pmatrix} = (136)(2875)$
        \item transpozicija\\
        \underline{primer:} $(i_1 i_2 \dots i_d) = (i_1 i_d)(i_1 i_{d-1}) \dots (i_1 i_2)$ 
    \end{enumerate}
\end{theorem}\

\noindent\textbf{Posledica 1.} \textit{Ciklus $\sigma$ je paran. $\iff$ Ciklus $\sigma$ je neparne dužine.} \\


\noindent\textbf{Posledica 2.} \textit{Neka je $\sigma = \tau_1 \circ \tau_2 \circ \dots \circ \tau_k$, gde su $\tau_i$ ciklusi dužine $t_i$. Tada je}\[ 
    |\sigma| = \operatorname{NZS}(t_1, t_2, \dots, t_k).
\]

\subsection{Definicija determinante (Lajbnicova formula)}
Determinanta $\operatorname{det}: \mathbb{M}_n(\mathbb{F}) \to \mathbb{F}$ je funkcija definisana formulom:
\[ \det \mathcal{A} = \sum_{\sigma \in \mathbb{S}_n} \left((-1)^{I(\sigma)} \prod_{i = 1}^{n}\alpha_{i \sigma(i)}\right), \]
gde je $\mathcal{A} \in \mathbb{M}_{n}(\mathbb{F})$ i $\sigma_i$ je $i$-ta po redu permutacija skupa $\mathbb{S}_n$ u leksikografskom poretku. Ova definicija 
ima prvenstveno teorijski značaj, jer je računski složena. Determinanta matrice $\mathcal{A}$ označava se sa $|\mathcal{A}|$\\

\noindent\underline{\textit{Intuicija/Izvođenje:}} Definišemo funkciju $n$-dimenzione zapremine, 
\[\operatorname{vol}_n: \underbrace{\mathbb{R}^n \times \dots \times \mathbb{R}^n}_n \to \mathbb{R}, \]
koja ima sledeća svojstva:
\begin{enumerate}
    \item linearnost po svakoj komponenti
    \item antisimetričnost (ako permutujemo dve komponente, znak se menja; iz ovoga sledi da ako su dve komponente iste, vrednost funkcije je 0)
    \item $\operatorname{vol}_n(e_1, e_2, \dots, e_n) = 1$
\end{enumerate}
Računamo:
\[
    \begin{aligned}
        &\operatorname{vol}_n \left(\begin{bmatrix}
        a_{11} \\ \vdots \\ a_{n1}
        \end{bmatrix}, 
        \underbrace{
        \begin{bmatrix}
            a_{12} \\ \vdots \\ a_{n2}
        \end{bmatrix}, 
        \dots,
        \begin{bmatrix}
            a_{1n} \\ \vdots \\ a_{nn}
        \end{bmatrix}}_{(\star)}
        \right) = \operatorname{vol}_n \bigg( a_{11}e_1 + \dots + a_{n1}e_n, \;(\star)\; \bigg) \\
        &= \sum_{j_1 = 1}^{n}a_{j_11} \operatorname{vol}_n(e_{j_1}, (\star)) 
        = \sum_{j_1 = 1}^{n}a_{j_11} \operatorname{vol}_n\left(e_{j_1}, \begin{bmatrix}
            a_{12} \\ \vdots \\ a_{n2}
        \end{bmatrix}, 
        \dots,
        \begin{bmatrix}
            a_{1n} \\ \vdots \\ a_{nn}
        \end{bmatrix}\right) \\
        &= \sum_{j_1 = 1}^{n} \sum_{j_2=1}^{n} a_{j_11} a_{j_22} \operatorname{vol}_n\left(e_{j_1}, e_{j_2}, \begin{bmatrix}
            a_{13} \\ \vdots \\ a_{n3}
        \end{bmatrix}, 
        \dots,
        \begin{bmatrix}
            a_{1n} \\ \vdots \\ a_{nn}
        \end{bmatrix}\right) =
    \end{aligned}
\]
\[
        = \sum_{j_1 = 1}^{n} \sum_{j_2 = 1}^{n} \dots \sum_{j_n=1}^{n} a_{j_11} a_{j_22} \dots a_{j_nn} \underbrace{\operatorname{vol}_n (e_{j_1}, e_{j_2}, \dots, e_{j_n})}_{0 \text{ ako se 2 komp. podudaraju}}
\]
Pretpostavimo da važi $e_{j_1} \neq e_{j_2} \neq \dots \neq e_{j_n}$ (inače je ceo izraz jednak 0, tj. vektori su linearno zavisni).Tada izraz postaje:
\[
    \sum_{(j_1, \dots, j_n) \;\in \;\mathbb{S}_n } a_{j_11} a_{j_22} \dots a_{j_nn} \operatorname{vol}_n (e_{j_1}, e_{j_2}, \dots, e_{j_n}),
\]
a prema definiciji funkcije $\operatorname{vol}_n$ važi
\[
    \operatorname{vol}_n (e_{j_1}, e_{j_2}, \dots, e_{j_n}) = \operatorname{sgn}((j_1, \dots, j_n)) = 
    \begin{cases}
        +1, & I((j_1, \dots, j_n)) \equiv_2 0 \\
        -1, & I((j_1, \dots, j_n)) \equiv_2 1
    \end{cases}.
\] Dakle, 
\[ \operatorname{vol}_n (e_{j_1}, e_{j_2}, \dots, e_{j_n}) = (-1)^{I(\sigma)}, \;\; \sigma = (j_1, \dots, j_n) \]
Definišimo permutaciju $\sigma \in S_n$ kao \textbf{inverznu} permutaciju od $(j_1, j_2, \dots, j_n)$, tj.
\[
\sigma(j_i) = i, \qquad \text{za svako } i = 1, 2, \dots, n.
\]
Tada važi:
\[
a_{j_1 1} a_{j_2 2} \dots a_{j_n n}
=
a_{1\sigma(1)} a_{2\sigma(2)} \dots a_{n\sigma(n)},
\]
jer kada $i$ prolazi kroz $1, \dots, n$, tada $k = j_i$ takođe prolazi kroz sve brojeve od $1$ do $n$ (samo u drugačijem redosledu), a pri tome je $i = \sigma(k)$.
Takođe, pošto inverzna permutacija ima isti broj inverzija (tj. istu parnost) kao i polazna, važi:
\[
(-1)^{I((j_1, \dots, j_n))} = (-1)^{I(\sigma)}.
\]
Zamenom u prethodnu sumu (i zamenom oznake $a_{ij}$ sa $\alpha_{ij}$) dobijamo upravo polaznu Lajbnicovu formulu:
\[
\sum_{\sigma \in S_n} (-1)^{I(\sigma)} \alpha_{1\sigma(1)} \alpha_{2\sigma(2)} \dots \alpha_{n\sigma(n)}.
\]\\

Ako imamo 3-dimenzionu bazu $(e_i, e_j, e_k)$, vidimo da je ona pozitivno orijentisana akko je $(i, j, k) \in \mathbb{S}_n$ parna permutacija. Determinanta
matrice čije kolone čine negativno orijentisanu bazu je negativna. O ovome se može razmišljati na sledeći način: znamo da kvadratna matrica predstavlja 
transformaciju prostora. Ako transformacija "obrće" zapreminu, prirodno se nameće da je ta zapremina negativna (zamislimo $1 \times 1 \times 1$ kocku koja 
se smanjuje dok zapremina ne postane 0, a zatim nastavi, tj. počne da raste "naopako"). Ovo i vidimo u prethodnom izvođenju. Sabiramo zapreminu "deo po deo", 
a negativno orijentisani delovi se svode na oduzimanje.



\begin{theorem}
    Neka je $\mathcal{A} = (\alpha_{ij}) \in \mathbb{M}_n$. Tada važi:
    \begin{enumerate}
        \item Ako je neka vrsta ili kolona matrice $\mathcal{A}$ nula vektor, onda je $|\mathcal{A}| = 0$.\\
        \noindent\underline{\textit{Intuicija:}} Npr. ako $3\times 3$ matrica ima tačno 1 nula vektor, geomtrijski je to onda ravan, pa ima zapreminu 0.
        \item $|\mathcal{A}| = |\mathcal{A^T}|$
        \item Ako je $A$ trougaona matrica, tada je $|\mathcal{A}| = \alpha_{11}\alpha_{22}\dots \alpha_{nn}$.\\
        \noindent\underline{\textit{Intuicija:}} Trougaona matrica je zapravo primenjivanje smicanja na originalni kvadar, pa ne menja zapreminu. 2D slučaj
        ovoga je paralelogram.
    \end{enumerate} 
\end{theorem}

\subsection{Induktivna definicija determinante i Laplasov razvoj}
\begin{itemize}
    \item \textbf{Minora} $\Delta_{ij}(\mathcal{A})$ matrice $\mathcal{A} = (\alpha_{ij}) \in \mathbb{M}_n$ dobija se
    uklanjanjem $i$-te vrste i $j$-te kolone determinante:
    \[ \Delta_{ij}(\mathcal{A}) = \begin{vmatrix}
        \alpha_{11} & \dots & \alpha_{1 \; j-1} & \alpha_{1 \; j+1} & \dots & \alpha_{1n} \\
        \vdots &        \vdots & \vdots             & \vdots          & \vdots  & \vdots \\
        \alpha_{i-1 \; 1} & \dots & \alpha_{i-1 \; j-1} & \alpha_{i-1 \; j+1} & \dots & \alpha_{i-1 \; n}\\
        \alpha_{i+1 \; 1} & \dots & \alpha_{i+1 \; j-1} & \alpha_{i+1 \; j+1} & \dots & \alpha_{i+1 \; n}\\
        \vdots &        \vdots & \vdots             & \vdots          & \vdots  & \vdots \\
        \alpha_{n1} & \dots & \alpha_{n \; j-1} & \alpha_{n \; j+1} & \dots & \alpha_{nn}
    \end{vmatrix} \]
    \item \textbf{Algebarski komplement (kofaktor)} $A_{ij}$ elementa $\alpha_{ij}$:
    \[ A_{ij} = (-1)^{i+j} \Delta_{ij}(\mathcal{A}) \]
    \item \textbf{Adjugovana matrica} $\tilde{\mathcal{A}}$ matrice $\mathcal{A}$:
    \[ \tilde{\mathcal{A}} = \operatorname{adj} (\mathcal{A}) = \begin{bmatrix}
        A_{11} & A_{21} & \dots & A_{n1} \\
        A_{12} & A_{22} & \dots & A_{n2} \\
        \vdots & \vdots & \ddots & \vdots \\
        A_{1n} & A_{2n} & \dots & A_{nn}
    \end{bmatrix} = [A_{ij}]^\mathcal{T}  \]
\end{itemize}

\begin{theorem}
    \textbf{(Laplasov razvoj):} Neka je $\mathcal{A} \in \mathbb{M}_n(\mathbb{F})$. Tada za $\forall i, j \in \{1, 2, \dots, n\}$ važi:
    \[ \sum_{j=1}^{n} \alpha_{ij}A_{rj} = \delta_{ir}\det \mathcal{A} \]
    \[ \sum_{i=1}^{n} \alpha_{ij}A_{ir} = \delta_{jr}\det \mathcal{A}.  \]
    \[ \left( \iff 
        \mathcal{A} \cdot \tilde{\mathcal{A}} = \tilde{\mathcal{A}} \cdot \mathcal{A} = (\operatorname{det} (\mathcal{A})) \mathbb{I}_n
     \right)   \]
\end{theorem}

\section{Sistemi linearnih jednačina}

\subsection{Definicija sistema linearnih jednačina}

Neka je dat skup $F = \{f_1, \dots, f_n\} \subset \mathbb{F}[x_1, \dots, x_m]$ linearnih polinoma u promenljivim $x_1, \dots, x_m$. On definiše sledeći \textbf{sistem linearnih
jednačina}:
\[
    \begin{aligned}
        f_1(x_1, \dots, x_m) &= \alpha_{11}x_1 + \alpha_{12}x_2 + \dots + \alpha_{1m}x_m - \beta_1 = 0, \\
        f_2(x_1, \dots, x_m) &= \alpha_{21}x_1 + \alpha_{22}x_2 + \dots + \alpha_{2m}x_m - \beta_2 = 0, \\
        &\vdots \\
        f_n(x_1, \dots, x_m) &= \alpha_{n1}x_1 + \alpha_{n2}x_2 + \dots + \alpha_{nm}x_m - \beta_n = 0,
    \end{aligned}
\] koji kraće zapisujemo kao $\sum_{k=1}^{m} \alpha_{ik}x_k = \beta_i, \;\; i = 1, \dots, n$ ili u matričnom obliku:
\[ \mathcal{A} = \begin{bmatrix}
    \alpha_{11} & \alpha_{12} & \dots & \alpha_{1m} \\
    \vdots      & \vdots      & \ddots & \vdots \\
    \alpha_{n1} & \alpha_{n2} & \dots & \alpha_{nm}
\end{bmatrix}, \; X = \begin{bmatrix}
    x_1 \\ \vdots \\ x_m
\end{bmatrix}, \; b = \begin{bmatrix}
    \beta_1 \\ \vdots \\ \beta_n
\end{bmatrix}, \; \mathcal{A}X = b. \]
Matricu $\mathcal{A}$ nazivamo \textbf{matricom sistema}. Definišemo i \textbf{matricu sistema}:

\[
\mathcal{A}_p = [\,\mathcal{A} \mid b\,] = 
\left[
\begin{array}{cccc|c}
\alpha_{11} & \alpha_{12} & \cdots & \alpha_{1m} & \beta_1 \\
\vdots      & \vdots      & \ddots & \vdots      & \vdots  \\
\alpha_{n1} & \alpha_{n2} & \cdots & \alpha_{nm} & \beta_n 
\end{array}
\right].
\]
\textbf{Rešenje sistema} je svaka uređena $m$-torka $(\gamma_1, \dots, \gamma_m) \in \mathbb{F}^m$ koja zamenom $x_i = \gamma_i$ u sistemu daje identitete u $\mathbb{F}$.
Ako je $\mathcal{A}$ invertibilna, onda važi
\[ X = \mathcal{A}^{-1}b. \]

\subsection{Geometrijska interpretacija sistema linearnih jednačina}

Ako su $U$ i $V$ vektorski prostori nad $\mathbb{F}$ i $\dim U = m, \; \dim V = n$ i ako su $e$ i $f$ njihove baze, tada $\mathcal{A}$ predstavlja matricu linearnog
operatora $A: U \to V$. Dakle, rešavanje sistema odgovara pronalaženju vektora $x \in U$ td. $Ax = b$.

\begin{theorem}
    Neka su $U$ i $V$ vektorski prostori nad $\mathbb{F}$ i $\dim U = m, \; \dim V = n$. Tada važi:
    \[ \text{sistem ima rešenja} \iff b \in \operatorname{Im} A \iff r(\mathcal{A}) = r(\mathcal{A}_p) \]
\end{theorem}
\noindent Ekvivalencija prvog i trećeg iskaza je \textbf{Kroneker-Kapelijeva teorema}.\\


\noindent\textbf{Posledica.} 
\[ r(\mathcal{A}) = n \text{ (tj. $\operatorname{Im} A = V$)} \implies \text{sistem ima rešenja} \]\\

Sistem jednačina naziva se \textbf{Kramerov sistem} ako:
\begin{enumerate}
    \item $m=n$
    \item $r(\mathcal{A}) = n$
\end{enumerate}

\begin{theorem}
    Kramerov sistem jednačina uvek ima jedinstveno rešenje, koje je dato formulom
    \[ x_k = \frac{\mathcal{D}_k}{\mathcal{D}}, \; k = 1, \dots, n,  \]
    gde je $\mathcal{D} = \det \mathcal{A} = \det [a_1, \dots, a_n]$ determinanta matrice sistema i gde su \linebreak $\mathcal{D}_k = \det[a_1, \dots, a_{k-1}, b, a_{k+1}, \dots, a_n]$
    determinante matrica koje se dobijaju iz $\mathcal{A}$ zamenom njene $k$-te kolone vektorom $b$.
\end{theorem}

\subsection{Homogeni sistemi linearnih jednačina}

Za linearni sistem jednačina kažemo da je \textbf{homogen} ako je $b = 0$. Rešavanje homogenog sistema ekvivalentno je rešavanju jednačine $Ax = 0$, tj. pronalaženju
jezgra linearnog operatora $A$.

\begin{theorem}
    Neka je dat homogeni sistem $\mathcal{A}X = 0$ td. $\operatorname{r}(A) = r$ i $\operatorname{d}(A) = d = m-r$.
    \begin{enumerate}
        \item Skup rešenja hom. sistema je $\ker A$. Tada postoji baza $v = (v_1, \dots, v_d)$ od $\ker A$ td. je svako rešenje sistema oblika
        \[ x = \sum_{i=1}^{d} \lambda_i v_i, \; \lambda_i \in \mathbb{R} \]
        \item Homogeni sistem uvek ima bar trivijalno rešenje $X = 0$, pa je uvek konzistentan.
        \item Homogeni Kramerov sistem ima samo trivijalno rešenje.
        \item Homogeni sistem u kom je broj jednačina manji od broja nepoznatih uvek ima netrivijalno rešenje.
    \end{enumerate}
\end{theorem}\

\textbf{Fundamentalni skup rešenja} homogenog sistema je skup
\[ \{ W_1, \dots, W_d \}, \; W_i = v_i(e) \] i svaki vektor koji je rešenje sistema se može predstaviti kao linearna kombinacija njegovih elemenata.

\subsection{Nehomogeni sistemi linearnih jednačina}

\begin{theorem}
    Neka je $\mathcal{A}X = B \neq 0$ nehomogeni sistem i $r = \operatorname{r}(\mathcal{A}) = \operatorname{r}(\mathcal{A}_p)$ rang sistema i $\operatorname{d}(\mathcal{A})
    = d = m-r$. Tada je skup rešenja sistema $\mathcal{A}X = B$ linearna mnogostrukost, tj.
    \[ x = v_0 + \sum_{i=1}^{d} \lambda_i v_i, \] gde je $v_0$ neko partikularno rešenje sistema, $(v_1, \dots, v_d)$ baza od $\ker A$ i $\lambda_i \in \mathbb{F}$ slobodni
    parametri.
\end{theorem}

\noindent\textbf{Posledica.} Neka je $\mathcal{A}X = B$ lin. sistem jednačina nad $\mathbb{F}$. Tada skup rešenja ovog sistema može biti:
\begin{enumerate}
    \item ako je $\operatorname{char} \mathbb{F} = 0$ (beskonačno polje): prazan skup, jednočlan i beskonačan.
    \item ako je $\operatorname{char} \mathbb{F} = p$, $p$ je prost (polje sa $p$ elemenata): prazan skup, jednočlan i stepen broja $p$.
\end{enumerate}

\subsection{Gausova metoda eliminacije}

Gausova metoda podrazumeva primenu elementarnih transformacija na matricu sistema. Dimenzija skupa rešenja jednaka je broju slobodnih parametara koji se dobiju 
primenom Gausove metode. Ovo je najefikasnija metoda za rešavanje linearnih sistema.

\begin{theorem}
    Neka je $\mathcal{A} \in \mathbb{M}_{nm}(\mathbb{F}), \; \operatorname{r}(\mathcal{A}) = r$ i $X \in \mathbb{F}^n$ td. $\mathcal{A}X = B$ i neka je 
    \[
    \mathcal{A}'_p = 
    \left[
    \begin{array}{cc|c}
    \mathbb{I}_r & \mathcal{C} & B'_r \\
    \mathcal{O}_{n-r, r} & \mathcal{O}_{n-r, m-r} & B'_{n-r}
    \end{array}
    \right]
    \] transformisana proširena matrica sistema dobijena iz $\mathcal{A}_p$ primenom elementarnih transformacija. Tada:
    \begin{description}
        \item[(nk)] lin. sistem $\mathcal{A}X = B$ nema rešenja ako je $B_{n-r}' \neq 0$.
        \item[(k)] lin. sistem $\mathcal{A}X = B$ ima rešenja ako je $B_{n-r}' = 0$ i skup rešenja je dat formulom
        \[ X = W_0 + \underbrace{\lambda_1W_1 + \dots + \lambda_dW_d}_{W_i \;\in\; \ker \mathcal{A}}, \; \lambda_i \in \mathbb{F}, \]
        gde je $W_0$ neko partikularno, a $W_i$ fundamentalna rešenja pridruženog homogenog sistema.
    \end{description}
\end{theorem}

\section{Redukcija linearnih operatora na konačnodimenzionim prostorima}

\subsection{Redukcija linearnog operatora}
\begin{itemize}
    \item Redukcija linearnog operatora je postupak izbora pogodne baze u kojoj matrica operatora poprima najjednostavniji mogući oblik (dijagonalni ili Žordanov).
    \item Cilj je pojednostaviti računske operacije (stepenovanje, eksponencijacija, rešavanje sistema), otkriti geometrijsku strukturu prostora (invarijantne potprostore) i klasifikovati sve operatore do na sličnost.
    \item Motivacija potiče iz potrebe za rešavanjem sprega diferencijalnih jednačina, gde dijagonalizacija razdvaja spregnute promenljive u nezavisne jednačine.
\end{itemize}

\subsection{Invarijantni potprostori}

\begin{itemize}
    \item Neka je $A \in \operatorname{Hom} V$ i $L \subseteq V$. $L$ je \textbf{invarijantan potprostor} operatora $A$ ako je $A(L) \subseteq L$.
    \item Ako je $L$ invarijantan potprostor od $A$, onda je dobro definisana restrikcija operatora $A$ na $L$. Dakle, preslikavanje $A_{|L}:L \to L, \; A_{|L}x = Ax$ je linearni operator.
    \item Svaki linearni operator $A \in \operatorname{Hom}V$ ima \textbf{trivijalne} invarijantne potprostore, \linebreak $L = \{0\}$ i $L = V$. Ostale nazivamo \textbf{pravi} ili \textbf{netrivijalni} invarijantni potprostori operatora $A$.
\end{itemize}

\begin{theorem}
    Neka je $A \in \operatorname{Hom}V$ linearni operator. Tada važi:
    \begin{enumerate}
        \item $\ker A$ i $\operatorname{Im}A$ su invarijantni potprostori od $A$
        \item $\ker A^j \subseteq \ker A^{j+1}, \;\; \forall j \in \mathbb{N}$
        \item $\operatorname{Im}A^j \supseteq \operatorname{Im}A^{j+1}, \;\; \forall j \in \mathbb{N}$
    \end{enumerate}
\end{theorem}

\subsection{Dekompozabilni i ireducibilni operatori}

\begin{itemize}
    \item Ako $A \in \operatorname{Hom}V$ ima netrivijalni invarijantni potprostor, kažemo da je \textbf{reducibilan (svodljiv)} na $V$. U suprotnom, kažemo da je \textbf{ireducibilan (nesvodljiv)} na $V$.
    \item Za operator koji ima invarijantne netrivijalne potprostore $L$ i $M$ takve da je $V = L \oplus M$, kažemo da je \textbf{dekompozabilan}.
    \item Ako za svaki netrivijalni invarijantni potprostor $L$ postoji invarijantni potprostor $M$ takav da je $V = L \oplus M$, kažemo da je operator $A$ \textbf{potpuno reducibilan} ili \textbf{potpuno svodljiv}.
    \item Ako je operator potpuno reducibilan, tada je i dekompozabilan.
    \item Ako je $A$ dekompozabilan, onda je potpuno određen svojim restrikcijama $A_1 = A_{|L}$ i $A_2 = A_{|M}$ jer $(\forall 
    x \in V)(\exists_1 l \in L)(\exists_1 m \in M)\;x= l + m$, pa važi
    \[ A(x) = A(l + m) = A(l) + A(m) = A_1(l) + A_2(m) \]
    Kažemo da je $A$ \textbf{direktna suma} operatora $A_1$ i $A_2$ (pišemo $A = A_1 \oplus A_2$).
    \item Neka je $A$ reducibilan i $L$ njegov invarijantni potprostor, a $M$ neki direktan komplement od $L$. Odaberemo bazu 
    $e$ td. prvih $k = \dim L$ vektora čine bazu $e_L$, a ostalih $n-k$ bazu $e_M$. U bazi $e$, matrica operatora $A$ ima oblik
    \[ A(e) = \begin{bmatrix}
        A_1(e_L) & A_{12} \\
        0 & A_2
    \end{bmatrix}, \]
    gde je $A_1 = A_{|L}$.\\
    \noindent\underline{\textit{Intuicija:}} Kako je $L$ invarijantni potprostor i $\dim L = k$, matrica $A_1(e_L)$ će biti 
    dimenzija $k \times k$ (jer slika $L$ u $L$). Ovo znači da će se ispod $A_1(e_L)$ nalaziti nula-matrica dimenzija $(n-k) \times k$
    (jer $A_1(e_L)$ ne slika ništa van $L$). Kako nije dato da je $M$ invarijantni potprostor, druga kolona u opštem slučaju neće
    imati nula-matricu. Dakle, matrica $A_{12}$ slika $M$ u $L$ (jer u $e$ prvo ide $e_L$ pa $e_M$), pa mora biti dimenzija 
    $k \times (n-k)$. Preostali blok $A_2$ je dimenzija $(n-k)\times (n-k)$ i on opisuje kako se vektori iz $M$ projektuju 
    nazad u $M$. On nije restrikcija u $M$ (jer nije invarijantan), već samo jedan blok u matrici. Dijagram:
    \[
        \begin{array}{cc}
         & \begin{array}{cc}
           \text{kolone } L\;\;\;\;\;\;\;\;\;\;\;\;\;\;\; & \text{kolone } M 
         \end{array} \\[4pt]
        \begin{array}{c}
         \text{vrste } L \\
         \text{vrste } M 
        \end{array} &
        \left[\begin{array}{c|c}
        A_1[k \times k] & A_{12}[k \times (n-k)] \\ \hline
        0 [(n-k) \times k] & A_2[(n-k) \times (n-k)]
        \end{array}\right]
        \end{array}
    \]
    \item Ako je i $M$ invarijantan, tj. $A$ je dekompozabilan, onda je matrica $A(e)$ kvazidijagonalna:
    \[ A(e) = \begin{bmatrix}
        A_1(e_L) & 0 \\
        0 & A_2(e_M)
    \end{bmatrix}, \]
    gde je $A_1 = A_{|L}$ i $A_2 = A_{|M}$.
    \item Ako je $A$ barem reducibilan, onda važi
    \[ \det (A(e)) = \det(A_1(e_L)) \det (A_2). \]
\end{itemize}

\begin{theorem}
    \label{rlonkdp1}
    Neka je $A \in \operatorname{Hom}V$ potpuno reducibilan linearni operator. Tada postoje ne-nula potprostori $L_1, 
    \dots, L_m$ t.d. je \[ V = \bigoplus_{i=1}^m L_i, \] i pri tome su za svako $i \in \mathbb{N}_m$ restrikcije
    $A_i = A_{|L_i}$ ireducibilni linearni operatori na $L_i$.
\end{theorem}

\begin{itemize}
    \item Dakle, ako je, kao u teoremi \ref{rlonkdp1}, $V = \bigoplus^m_{i=1} L_i$ i ako su $A_i = A_{|L_i}$, tada je
    i $A = \bigoplus^m_{i=1} A_i$. Neka je $e = (e_{L_1}, \dots, e_{L_m})$ baza od $V$ t.d. su $e_{L_i}$ baze od $L_i$, onda
    je matrica operatora $A$ u toj bazi kvazidijagonalna, tj. oblika
    \[ A(e) = \begin{bmatrix}
        A_1(e_{L_1}) & 0 & \dots & 0 \\
        0 & A_2(e_{L_2}) & \dots & 0 \\
        \vdots & \vdots & \ddots & \vdots \\
        0 & 0 & \dots & A_m(e_{L_m})
    \end{bmatrix}. \]
    \item Tada važi i formula
    \[ \det (A(e)) = \prod^m_{i=1} \det(A_i(e_{L_i})). \]
\end{itemize}

\begin{theorem}
    \textbf{(Fitingova dekompozicija):} Neka je $A: V \to V$ linearni operator i $\dim V < \infty$. Tada postoje $m \in \mathbb{N}$
    i jedinstveni invarijantni potprostori $L = \ker A^m$ i $M = \operatorname{Im}A^m$ od $A$ t.d. je 
    \begin{enumerate}
        \item $V = L \oplus M$,
        \item $B = A_{|M}$ je regularan operator i $C = A_{|L}$ je nilpotentan operator.
    \end{enumerate}
\end{theorem}

\subsection{Karakteristični polinom i invarijante sličnosti}

\begin{itemize}
    \item Ako je $\mathcal{A} \in \mathbb{M}_n(\mathbb{F})$, tada matricu $\mathcal{C} = \mathcal{A} - \lambda\mathbb{I}_n$, 
    gde je $\lambda$ promenljiva, nazivamo \textbf{karakteristična matrica} od $\mathcal{A}$.
    \item Determinantu od $\mathcal{C}$, tj. polinom u $\lambda$:
    \[ \det(\mathcal{A} - \lambda \mathbb{I}_n) = \begin{vmatrix}
        \alpha_{11} - \lambda & \alpha_{12} & \dots & \alpha_{1n}\\
        \alpha_{21} & \alpha_{22}-\lambda & \dots & \alpha_{2n} \\
        \vdots & \vdots & \ddots & \vdots \\
        \alpha_{n1} & \alpha{n2} & \dots & \alpha_{nn} - \lambda
    \end{vmatrix} = \kappa_\mathcal{A}(\lambda)\]
    nazivamo \textbf{karakteristični polinom} $\kappa_\mathcal{A}(\lambda)$ matrice $\mathcal{A}$.
    \item \textbf{Karakteristična jednačina} matrice $\mathcal{A}$: $\kappa_\mathcal{A}(\lambda) = 0$
    \item Važi jednakost:
    \[ \kappa_\mathcal{A}(\lambda) = \prod^{n}_{i = 1} (\alpha_{ii} - \lambda) + q(\lambda), \;\; \deg q \leq n-2 \]
    \item \textbf{Invarijante sličnosti} su svojstva linearnog operatora $A \in \operatorname{Hom}V$ koja ne zavise od baze.
\end{itemize}

\begin{theorem}
    Karakteristični polinom je invarijanta sličnosti.
\end{theorem}\

\noindent\textbf{Posledica 1.} \textit{Trag matrice je invarijanta sličnosti.}\\

\noindent\textbf{Posledica 2.} \textit{Determinanta matrice je invarijanta sličnosti.}

\subsection{Hamilton-Kejlijeva teorema}

\begin{theorem}
    \textbf{(Hamilton-Kejli):} Neka je $\mathcal{A} \in \mathbb{M}_n$. Tada je \[ \kappa_\mathcal{A}(\mathcal{A}) = 0, \]
    tj. svaka matrica poništava svoj karakteristični polinom.
\end{theorem}

\subsection{Minimalni polinom matrice}

\begin{itemize}
    \item \textbf{Minimalni polinom} matrice $\mathcal{A}$ je monični polinom $\mu_\mathcal{A}(\lambda) \in \mathbb{F}[\lambda]$ 
    najmanjeg stepena za koji važi \[ \mu_\mathcal{A}(\mathcal{A})=0. \]
\end{itemize}

\begin{theorem}
    Neka je $\mathcal{A}\in \mathbb{M}_n(\mathbb{F})$.
    \begin{enumerate}
        \item Ako je $p \in \mathbb{F}[\lambda]$ takav da je $p(\mathcal{A}) = 0$, tada $\mu_\mathcal{A}(\lambda)\;|\;p(\lambda)$.
        Specijalno, važi \[\mu_\mathcal{A}(\lambda)\;|\;\kappa_\mathcal{A}(\lambda).\]
        \item Minimalni polinom je invarijanta sličnosti.
    \end{enumerate}
\end{theorem}

\begin{theorem}
    Neka je $p(\lambda) = \lambda^n + \alpha_{n-1}\lambda^n + \dots + \alpha_1\lambda + \alpha_0 \in \mathbb{F}[\lambda], \; n > 0$.
    Tada je $p(\lambda)$ jednak minimalnom polinomu matrice
    \[\mathcal{A} = A(e) = \begin{bmatrix}
        0 & 0 & \dots & 0 &-\alpha_0 \\
        1 & 0 & \dots & 0 &-\alpha_1 \\
        0 & 1 & \dots & 0 &-\alpha_2\\
        \vdots & \vdots & \ddots & \vdots & \vdots \\
        0 & 0 & \dots & 1 & -\alpha_{n-1}
    \end{bmatrix}.\]
    Takođe važi $\kappa_\mathcal{A}(\lambda) = (-1)^n\mu_\mathcal{A}(\lambda)$.
\end{theorem}

\begin{theorem}
    Neka je $\mathcal{A} \in \mathbb{M}_n(\mathbb{F})$ i $p(\lambda) = \alpha_k\lambda^k + \dots + \alpha_1\lambda + \alpha_0
     \in \mathbb{F}[\lambda]$ t.d. $p(\mathcal{A})=0$. Tada važi
    \begin{enumerate}
        \item $\alpha_0 \neq 0 \implies \mathcal{A}$ je invertibilna,
        \item ako je $p = \mu_\mathcal{A}$, tada važi
        \[  \alpha_0 \neq 0 \iff \mathcal{A} \text{ je invertibilna.}\]
    \end{enumerate}
    Ukoliko je $\mathcal{A}$ invertibilina, njen inverz se pronalazi po formuli
    \[ \mathcal{A}^{-1} = - \frac{1}{\alpha_0}\left( \alpha_k\mathcal{A}^{k-1}+\dots + \alpha_1\mathbb{I}_n \right). \]
\end{theorem}

\subsection{Sopstvene vrednosti linearnog operatora}

\begin{itemize}
    \item Skalar $\lambda \in \mathbb{F}$ se zove \textbf{sopstvena (svojstvena, karakteristična) vrednost} linearnog operatora $A \in \operatorname{Hom}V$ ako
    \[ (\exists x \in V)\; Ax = \lambda x \land x \neq 0. \] Vektor $x$ se zove \textbf{sopstveni (svojstveni, karakteristični) vektor} koji odgovara 
    sopstvenoj vrednosti $\lambda$.
    \item \textbf{Spektar} $\operatorname{Sp}(A)$ linearnog operatora $A$ - skup svih sopstvenih vrednosti od $A$.
    \item \textbf{Sopstveni potprostor sopstvene vrednosti} $\lambda$ operatora $A$:
    \[ V_\lambda^A = V_\lambda = \{x \in V \;|\; Ax = \lambda x\} \]
\end{itemize}

\begin{theorem}
    $V_\lambda^A$ je potprostor od $V$.
\end{theorem}

\begin{theorem}
    Neka je $A \in \operatorname{Hom}V$ linearni operator. Tada važi
    \[ \lambda_0 \in \operatorname{Sp}(A) \iff \kappa_A(\lambda_0) = 0 \]
\end{theorem}

\noindent\textbf{Posledica.} \textit{Sopstvena vrednost je invarijanta sličnosti.}

\begin{theorem}
    Linearni operator koji deluje na $n$-dimenzionom vektorskom prostoru nad algebarski zatvorenim poljem ima tačno $n$ sopstvenih vrednosti (među kojima može
    biti istih).
\end{theorem}

\begin{theorem}
    Neka je $\mathcal{A} \in \mathbb{M}_n(\mathbb{F})$. Tada:
    \begin{enumerate}
        \item Svaki ireducibilni faktor karakterističnog polinoma matrice $\mathcal{A}$ ujedno je i ireducibilni faktor minimalnog polinoma iste.
        \item $\kappa_\mathcal{A}(\lambda)$ nema višestrukih nula $\implies \kappa_\mathcal{A}(\lambda) = \mu_\mathcal{A}(\lambda)$
    \end{enumerate}
\end{theorem}

\subsection{Dijagonalizacija linearnog operatora}

\begin{itemize}
    \item Ako operator $A \in \operatorname{Hom}V$ ima $\dim V$ jednodimenzionih invarijantnih potprostora, tada postoji baza $f$ od $V$, u kojoj je matrica operatora $A(f)$ dijagonalna.
    \item Proces pronalaženja baze (ako ona postoji) u kojoj je matrica operatora $A$ dijagonalna, zove se \textbf{dijagonalizacija} linearnog operatora.
    \item Za operator $A$ kažemo da je \textbf{dijagonalizabilan} ako postoji baza $f$ u kojoj je matrica operatora $A(f)$ dijagonalna.
\end{itemize}

\begin{lemma}
    Neka su $\lambda_1, \dots, \lambda_k$ međusobno različite sopstvene vrednosti linearnog operatora $A \in \operatorname{Hom}V$ i 
    neka su redom $v_1, \dots, v_k$ odgovarajući sopstveni vektori. Tada je skup vektora $\{v_1, \dots, v_k\}$ linearno nezavisan.
\end{lemma}

\begin{theorem}
    Neka je $A \in \operatorname{Hom}V$ linearni operator nad $\mathbb{F}$. Tada su sledeći iskazi ekvivalentni:
    \begin{enumerate}
        \item $A$ je dijagonalizabilan.
        \item Postoji baza $e$ od $V$ koja se sastoji od sopstvenih vektora.
        \item $\mu_\mathcal{A}(\lambda)$ je proizvod međusobno različitih linearnih faktora oblika $\lambda - \lambda_i, \; \lambda_i \in \mathbb{F}$.
    \end{enumerate}
\end{theorem}

\noindent\textbf{Posledica 1.} \textit{Ako je operator $A$ dijagonalizabilan, tada se na glavnoj dijagonali njegove matice u 
bazi u kojoj je predstavljen dijagonalnom matricom nalaze sopstvene vrednosti operatora $A$. Dakle, oblika je}
\[ \mathcal{A}(e) = \begin{bmatrix}
    \lambda_1 & 0 & \dots & 0 \\
    0 & \lambda_2 & \dots & 0 \\
    \vdots & \vdots & \ddots & \vdots \\
    0 & 0 & \dots & \lambda_n
\end{bmatrix}. \]

\noindent\textbf{Posledica 2.} \textit{Ako operator $A$ ima $n = \dim V$ različitih sopstvenih vrednosti, onda je on dijagonalizabilan.}
\newpage
\begin{algorithm}[!h]
    \caption{Dijagonalizacija matrice}
    \begin{enumerate}
        \item Naći sve sopstvene vrednosti operatora $A$.
        \item Formirati polinom 
        \[ p(\lambda) = \prod_{i=1}^{k}(\lambda - \lambda_i), \;\; \lambda_i \in \operatorname{Sp}(A). \]
        \item Proveriti da li je matrica dijagonalizabilna:
        \[ p(\mathcal{A}) = 0 \iff \mathcal{A} \text{ je dijagonalizabilna.} \]
        Ako nije dijagonalizabilna, algoritam se zaustavlja.
        \item Naći sopstvene vektore u polaznoj bazi $e$, tj. vektore 
        \[v_i(e) = (t_{1i}, \dots, t_{ni}), \;\; i \in \mathbb{N}_n\]
        i formirati matricu prelaska $T_{ev}$.
        \item Ako operator $A$ u bazi $e$ ima matricu $\mathcal{A}(e)$, matrica
        \[ \mathcal{A}(v) = T_{ev}^{-1}\mathcal{A}(e)T_{ev} \] je dijagonalna.
    \end{enumerate}
\end{algorithm}

\begin{itemize}
    \item Neka je $A \in \operatorname{Hom}V$ linearni operator i $\lambda_0 \in \operatorname{Sp}(A)$. Tada definišemo:
    \begin{enumerate}
        \item \textbf{Algebarska višestrukost} sopstvene vrednosti $\lambda_0$ je višestrukost $\lambda_0$ kao nule karakterističnog
        polinoma $\kappa_{A}(\lambda)$.
        \item \textbf{Geometrijska višestrukost} sopstvene vrednosti $\lambda_0$ je dimenzija njenog sopstvenog prostora $V_{\lambda_0}$.
    \end{enumerate}
\end{itemize}

\begin{theorem}
    Neka je $A \in \operatorname{Hom}V$ linearni operator i $\lambda_0$ neka njegova sopstvena vrednost. Algebarska višestrukost
    sopstvene vrednosti $\lambda_0$ veća je ili jednaka od njene geometrijske višestrukosti.
\end{theorem}

\subsection{Ireducibilni faktori karakterističnog polinoma i invarijantni potprostori}

\begin{theorem}
    \label{irfkpiip1}
    Neka je $A \in \operatorname{Hom}V$ linearni operator i neka su $f, g, h \in \mathbb{F}[x]$ polinomi t.d.
    \begin{enumerate}
        \item $f = gh$,
        \item $f(A) = 0$,
        \item $g$ i $h$ su uzajamno prosti polinomi.
    \end{enumerate}
    Tada je $V = \ker g(A) \oplus \ker h(A)$.
\end{theorem}
\noindent\underline{\textit{Intuicija:}} Ako polinom $f$ poništava operator $A$ i ako $f$ rastavimo na 2 uzajamno prosta faktora $g$ i $h$, tada se ceo
prostor $V$ raspada na direktnu sumu jezgara ta 2 faktora. Dakle, praktično razdvajamo $V$ na dva dela. U prvom ($\ker g(A)$) važi $g(A) = 0$, a u drugom 
($\ker h(A)$) važi $h(A) = 0$. Ova teorema nam garantuje da se ti delovi ne poklapaju i da pokrivaju ceo $V$.\\

\noindent\textbf{Posledica.} \textit{Neka je $A \in \operatorname{Hom}V$ linearni operator i neka su $f, g, h \in \mathbb{F}[x]$ polinomi t.d.
    \begin{enumerate}
        \item $f = gh$,
        \item $f(A) = 0$,
        \item $g$ i $h$ su uzajamno prosti polinomi,
        \item $f$ je minimalan polinom od $A$,
        \item $g$ i $h$ su monični polinomi.
    \end{enumerate}
    Tada su $g$ i $h$ minimalni polinomi restrikcija $A_{|\ker g(A)}$ i $A_{| \ker h(A)}$ respektivno.
}

\begin{theorem}
    Neka je $A \in \operatorname{Hom}V$ linearni operator čiji je minimalni polinom
    \[ \mu_A = \prod_{i=1}^{k}p_i^{s_i}, \]
    gde su $p_i$ različiti monični polinomi. Tada je $V = \bigoplus_{i=1}^k L_i$, pri čemu su $L_i = \ker p_i^{s_i}$ invarijantni potprostori od $A$, 
    i $p_i^{s_i}$ je minimalni polinom restrikcije $A_{|L_i}$ na $L_i, \; i \in \mathbb{N}_k$.
\end{theorem}
\noindent\underline{\textit{Intuicija:}} Ovo je samo primena teoreme \ref{irfkpiip1} maksimalan broj puta. \\
Dakle, matrica operatora $A$ u bazi koja prati ovu dekompoziciju izgleda ovako:
\[ \mathcal{A}(e) = \begin{bmatrix}
    A_1 & 0 & \dots & 0 \\
    0 & A_2 & \dots & 0 \\
    \vdots & \vdots & \ddots & \vdots \\
    0 & 0 & \dots & A_k
\end{bmatrix}. \]


\subsection{Nilpotentni operatori}

\begin{itemize}
    \item Za operator $A \in \operatorname{Hom}V$ kažemo da je \textbf{nilpotentan} ako postoji $k \in \mathbb{N}$ t.d. je $A^k = 0$.
    \item \textbf{Indeks nilpotentnosti} $k$ operatora $A$: \[k = \min \{k \in \mathbb{N}\;|\;A^k = 0 \land A^{k-1}\neq 0\}\]
    \item Ako je $A$ nilpotentni operator indeksa nilpotentnosti $k$, njegov karakteristični polinom je $\kappa_A(\lambda) = (-1)^n\lambda^n$, a
    minimalni $\mu_A(\lambda) = \lambda^k$, pa mu je jedina sopstvena vrednost $\lambda = 0$. Takođe, $\operatorname{Tr}(A) = \det A = 0$ i $k \leq \dim V$.
    \item \textbf{Kanonska nilpotentna matrica indeksa $k$}:
    \[ \mathcal{N}_k = \begin{bmatrix}
        0 & 1 & 0 & \dots & 0 \\
        0 & 0 & 1 & \dots & 0 \\
        \vdots & \vdots & \vdots & \ddots & \vdots \\
        0 & 0 & 0 & \dots & 1 \\
        0 & 0 & 0 & \dots & 0
    \end{bmatrix}_{k \times k} \]
\end{itemize}

\begin{theorem}
    Neka je $A \in \operatorname{Hom}V$ nilpotentni operator indeksa nilpotentnosti $k$. Tada:
    \begin{enumerate}
        \item Postoji vektor $e \in V$ t.d. je skup $S_1 = \{e, Ae, \dots, A^{k-1}e\}$ linearno nezavisan;
        \item $L_1 = \mathcal{L}(S_1)$ je invarijantan potprostor od $A$;
        \item Restrikcija $A_{|L_1}$ je nilpotentan operator indeksa nilpotentnosti $k$, čija je matrica u bazi $e_{L_1} = (A^{k-1}e, \dots, Ae, e)$ 
        kanonska nilpotentna matrica indeksa nilpotentnosti $k$, tj. oblika
        \[ \mathcal{A}_{|L_1}(e_{L_1}) = \begin{bmatrix}
        0 & 1 & 0 & \dots & 0 \\
        0 & 0 & 1 & \dots & 0 \\
        \vdots & \vdots & \vdots & \ddots & \vdots \\
        0 & 0 & 0 & \dots & 1 \\
        0 & 0 & 0 & \dots & 0
    \end{bmatrix}_{k \times k} \]
    \end{enumerate}
\end{theorem}

\begin{itemize}
    \item \textbf{Ciklički operator} - linearni operator za koji postoji baza $\mathcal{B} = (e, Ae, \dots, A^{k-1}e)$ t.d. je $A^ke \in \mathcal{L}(\mathcal{B})$.
    \item Nilpotentan ciklički operator $A$ u bazi $e_{L_1} = (A^{k-1}e, \dots, Ae, e)$ ima kanonsku nilpotentnu matricu indeksa nilpotentnosti $k$, tj. oblika
    \[ \mathcal{A}(e_{L_1}) = \begin{bmatrix}
        0 & 1 & 0 & \dots & 0 \\
        0 & 0 & 1 & \dots & 0 \\
        \vdots & \vdots & \vdots & \ddots & \vdots \\
        0 & 0 & 0 & \dots & 1 \\
        0 & 0 & 0 & \dots & 0
    \end{bmatrix}_{k \times k} \]
    \item Baza $e_{L_1} = (A^{k-1}e, \dots, Ae, e)$  zove se \textbf{ciklička baza} od $A_1$, a $L_1 = \mathcal{L}(\{e, Ae, \dots, A^{k-1}e\})$ se zove 
    \textbf{ciklički potprostor} operatora $A$.
\end{itemize}

\subsection{Fermaova, Rolova, Košijeva i Lagranžova teorema}

\noindent\textbf{Fermaova teorema}: Neka je $U_\delta = (c - \delta,c + \delta) \;\delta$-okolina tačke $c \in \mathbb{R}$ i funkcija $f:U_\delta \xrightarrow{}\mathbb{R}$ diferencijabilna u tački $c$. Ako funkcija $f$ u tački $c$ ima lokalni ekstremum, onda je $f'(c) = 0.$\\

\noindent\textbf{Rolova teorema}: Neka je funkcija $f:[a,b] \xrightarrow{} \mathbb{R}$:
\begin{itemize}
    \item neprekidna u svim tačkama segmenta $[a,b]$,
    \item diferencijabilna u svim tačkama intervala $(a,b)$,
    \item ima jednake vrednosti na krajevima tog segmenta: $f(a) = f(b)$.
\end{itemize}
Tada postoji tačka $c \in (a,b)$ td. $f'(c) = 0$.\\

    \centering
    \includegraphics[width=0.4\textwidth]{slike/rolle.pdf}
    \caption{Rolova teorema}
\end{figure}

\noindent \textbf{Košijeva teorema}: Neka su funkcije $f, g: [a, b] \to \mathbb{R}, \; (g'(x) \neq 0$ za $x \in (a, b))$:
\begin{itemize}
    \item neprekidne na $[a, b]$
    \item diferencijabilne na $(a, b)$
\end{itemize}
Tada postoji tačka $x_0 \in (a, b)$ t.d. važi
$ \displaystyle \frac{f(b) - f(a)}{g(b) - g(a)} = \frac{f'(x_0)}{g'(x_0)} $.

\noindent\textbf{Lagranžova teorema}: Neka je funkcija $f:[a,b] \xrightarrow{} \mathbb{R}$:
\begin{itemize}
    \item neprekidna u svim tačkama segmenta $[a,b]$,
    \item diferencijabilna u svim tačkama intervala $(a,b)$.
\end{itemize}
Tada postoji tačka $c \in (a,b)$ td. $f'(c)(b - a) = f(b) - f(a)$.
